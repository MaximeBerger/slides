\textbf{\large Des formules de quadrature aux schémas d'EDO}

\medskip

On considère le problème de Cauchy
\[
  y'(t) = f(t,y(t)), \qquad y(t_0) = y_0.
\]

Pour résoudre ce problème de manière approchée la clé est de l'écrire en intégrant entre $t_n$ et $t_{n+1} = t_n + h$ :
\[
  y(t_{n+1}) - y(t_n) = \int_{t_n}^{t_{n+1}} f(t,y(t))\,dt.
\]

Maintenant, c'est l'intégrale qu'il faut traiter. Chaque manière d'approcher cette intégrale donne un schéma numérique différent.

\medskip

\begin{tabular}{|m{0.20\linewidth}|m{0.50\linewidth}|m{0.17\linewidth}|}
  \hline
  \textbf{Quadrature} & \textbf{Approximation} & \textbf{Schéma} \\[1em]
  \hline
  Rectangles à gauche &
    $\displaystyle  y(t_{n+1}) - y(t_n) = h\,f(t_n, y_n)$ &
    Euler explicite \\[1em]
  \hline
  Rectangles à droite &
    $\displaystyle  y(t_{n+1}) - y(t_n) = h\,f(t_{n+1}, y_{n+1})$ &
    Euler implicite \\[1em]
  \hline
  Trapèzes &
    $\displaystyle  y(t_{n+1}) - y(t_n) = \frac{h}{2}\bigl[f(t_n, y_n) + f(t_{n+1}, y_{n+1})\bigr]$ &
    Crank--Nicolson \\[1em]
  \hline
   Lagrange d'ordre 2 &
    voir exercice 1 &
    Runge Kutta\\[1em]
  \hline
\end{tabular}

\medskip

\section*{Exercice 1 --- Euler explicite : précision et stabilité}
% ====================================================================

On considère le problème de Cauchy
\[
  y'(t) = -2\,y(t), \qquad y(0) = 1.
\]
La solution exacte est $y(t) = e^{-2t}$. On note $h = t_{n+1} - t_n$ le pas de temps.

\subsection*{1) Euler explicite}
\begin{enumerate}[itemsep=0.5em]

\item Écrire la formule d'Euler explicite.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    L'Euler explicite donne
    \[
      y_{n+1} = y_n + h\,f(t_n, y_n)
               = y_n + h\,(-2y_n)
               = (1 - 2h)\,y_n.
    \]
  \end{solution}
}{}

\item Calculer les approximations $y_1$ et $y_2$ pour $h = 0{,}25$.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Avec $h = 0{,}25$ on a $1 - 2h = 0{,}5$.
    \[
      y_1 = 0{,}5 \times y_0 = 0{,}5,
      \qquad
      y_2 = 0{,}5 \times y_1 = 0{,}25.
    \]
  \end{solution}
}{}

\item Donner la valeur exacte de $y(0{,}5)$, puis l'erreur absolue commise par $y_2$.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    La valeur exacte est $y(0{,}5) = e^{-1} \approx 0{,}3679$.

    L'erreur absolue vaut $|0{,}3679 - 0{,}25| \approx 0{,}118$.
  \end{solution}
}{}

\item Refaire le calcul avec $h = 0{,}125$ (4 pas pour atteindre $t = 0{,}5$). En mutlipliant le pas par 2, par combien est divisée l'erreur ? \newline
La méthode est dite \textit{d'ordre 1} en $h$.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Avec $h = 0{,}125$, $1 - 2h = 0{,}75$.
    \[
      y_1 = 0{,}75, \quad
      y_2 = 0{,}5625, \quad
      y_3 \approx 0{,}4219, \quad
      y_4 \approx 0{,}3164.
    \]
    L'erreur est $|0{,}3679 - 0{,}3164| \approx 0{,}0515$.
    En divisant le pas par $2$, l'erreur est passée de $0{,}118$ à $0{,}052$ : elle est bien environ divisée par $2$.
  \end{solution}
}{}

\item Que devient l'algorithme si on choisit $h = 1$ ? Commenter.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Avec $h = 1$, on a $1 - 2h = -1$. Donc $y_{n+1} = -y_n$ :
    \[
      y_0 = 1, \quad y_1 = -1, \quad y_2 = 1, \quad \ldots
    \]
    La suite oscille entre $1$ et $-1$. Elle ne converge pas vers $0$, contrairement à la solution exacte qui décroît exponentiellement.
    Le pas est trop grand : il faut $|1 - 2h| < 1$, soit $h < 1$, pour que le schéma soit stable.
  \end{solution}
}{}

\end{enumerate}


\subsection*{2) Euler implicite}
\begin{enumerate}[itemsep=0.5em]

\item
En approchant l'intégrale $\displaystyle\int_{t_n}^{t_{n+1}} f(t,y)\,dt$ par la méthode des \emph{rectangles à droite}, écrire le schéma d'Euler implicite.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Les rectangles à droite donnent $\int_{t_n}^{t_{n+1}} f \approx h\,f(t_{n+1}, y_{n+1})$. Donc :
    \[
      y_{n+1} = y_n + h\,(-2\,y_{n+1}).
    \]
  \end{solution}
}{}

\item Déterminer l'expression de $y_{n+1}$ en fonction de $y_n$ et $h$.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    $y_{n+1} + 2h\,y_{n+1} = y_n$, donc $(1 + 2h)\,y_{n+1} = y_n$, d'où
    $y_{n+1} = \dfrac{y_n}{1 + 2h}$.
  \end{solution}
}{}

\item Calculer $y_1$ et $y_2$ avec $h = 1$. Comparer avec l'Euler explicite sur le même pas. Existe-t-il des valeurs de $h$ qui font diverger la méthode ? ? 

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Avec $h = 1$ : $y_{n+1} = \frac{1}{3}\,y_n$.
    \[
      y_1 = \frac{1}{3} \approx 0{,}333,
      \qquad
      y_2 = \frac{1}{9} \approx 0{,}111.
    \]
    Contrairement à l'explicite (qui oscillait avec $h = 1$),
    l'implicite reste stable et décroissant.

    Pour tout $h > 0$, la méthode est stable.
  \end{solution}
}{}

\end{enumerate}

\subsection*{3) Crank--Nicolson}
\begin{enumerate}[itemsep=0.5em]

\item
En utilisant maintenant la méthode des \emph{trapèzes} pour estimer l'intégrale, exprimer $y_{n+1}$ en fonction de $y_n$ et $h$.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    La formule des trapèzes donne
    \[
      y_{n+1} = y_n + \frac{h}{2}\bigl[(-2y_n) + (-2y_{n+1})\bigr]
              = y_n - h\,y_n - h\,y_{n+1}.
    \]
    En regroupant :
    $(1 + h)\,y_{n+1} = (1 - h)\,y_n$, d'où
    $y_{n+1} = \dfrac{1-h}{1+h}\,y_n$.
  \end{solution}
}{}

\item Calculer $y_1$ et $y_2$ pour $h = 0{,}5$ avec chacune des trois méthodes, puis comparer avec la solution exacte.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Avec $h = 0{,}5$, on cherche $y_1 \approx y(0{,}5) = e^{-1} \approx 0{,}3679$ et $y_2 \approx y(1) = e^{-2} \approx 0{,}1353$.

    \medskip

    \begin{tabular}{lccc}
      & $y_1$ & $y_2$ & Facteur de récurrence \\
      \hline
      Euler explicite  & $0$ & $0$ & $1 - 2 \times 0{,}5 = 0$ \\
      Euler implicite  & $0{,}5$ & $0{,}25$ & $1/(1+1) = 0{,}5$ \\
      Crank--Nicolson  & $1/3 \approx 0{,}333$ & $1/9 \approx 0{,}111$ & $(1-0{,}5)/(1+0{,}5) = 1/3$ \\
      Exact            & $0{,}368$ & $0{,}135$ &
    \end{tabular}

    \medskip

    L'explicite s'annule (c'est un cas limite de stabilité).
    L'implicite sous-estime la décroissance.
    Crank--Nicolson est le plus proche de la solution exacte : c'est une méthode d'ordre $2$.
  \end{solution}
}{}

\item Pour chaque schéma, indiquer la condition sur $h$ qui assure la stabilité
(i.e. $|y_{n+1}/y_n| < 1$). Lequel est le plus robuste ?

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    \begin{itemize}
      \item Explicite : $|1 - 2h| < 1 \Leftrightarrow 0 < h < 1$.
      \item Implicite : $\dfrac{1}{1+2h} < 1$ pour tout $h > 0$. Toujours stable.
      \item Crank--Nicolson : $\left|\dfrac{1-h}{1+h}\right| < 1$ pour tout $h > 0$. Toujours stable.
    \end{itemize}
    Les schémas implicite et Crank--Nicolson sont \textbf{inconditionnellement stables}.
    L'implicite est le plus robuste (pas d'oscillation possible),
    Crank--Nicolson combine stabilité et ordre $2$.
  \end{solution}
}{}

\end{enumerate}

\subsection*{4) Runge--Kutta}
Vers une méthode d'ordre plus élevé.
On repart de
\[
  y(t_{n+1})-y(t_n)=\int_{t_n}^{t_{n+1}} g(t)\,dt,
  \qquad g(t)=f(t,y(t)).
\]
L'objectif est de mieux approcher l'intégrale en utilisant trois points
$t_n$, $t_n+\frac h2$, $t_{n+1}$.
\begin{enumerate}[label=\alph*)]
  \item Pour simplifier les calculs, on pose $x = \frac{t-t_n}{h} \in [0,1]$. \newline 
  Quand $t=t_n$, combien vaut $x$ ? et si  $t = t_n+\frac h2$ et si $t= t_{n+1}$ ?

  \item Construire le polynôme de Lagrange d'ordre $2$, noté $P_2$, qui interpole
  $g$ en ces trois noeuds. On pourra les ecrire avec la variable $x$.

  \item Intégrer $P_2$ sur $[t_n,t_{n+1}]$ et montrer que
  \[
    y_{n+1}=y_n+\frac h6\Bigl(g(t_n)+4g(t_n+\tfrac h2)+g(t_{n+1})\Bigr).
  \]
  C'est la formule de Simpson appliquée à $g=f(t,y(t))$.
  \item Pour obtenir une méthode explicite simple à appliquer, on introduit
  quatre pentes calculées entre $t_n$ et $t_{n+1}$ :
  \[
    k_1=f(t_n,y_n),\qquad
    k_2=f\!\Bigl(t_n+\frac h2,\,y_n+\frac h2k_1\Bigr),
  \]
  \[
    k_3=f\!\Bigl(t_n+\frac h2,\,y_n+\frac h2k_2\Bigr),\qquad
    k_4=f(t_n+h,\,y_n+hk_3).
  \]
  Interpréter le rôle de chaque pente (début, milieu, milieu corrigé, fin).
  \item Écrire alors la mise à jour de $y_{n+1}$ en combinant ces pentes :
  \[
    y_{n+1}=y_n+\frac h6\,(k_1+2k_2+2k_3+k_4).
  \]
\end{enumerate}

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    En posant $x=\frac{t-t_n}{h}\in[0,1]$, les trois noeuds sont $x=0,\frac12,1$.
    Les polynômes de base de Lagrange sont
    \[
      \ell_0(x)=2(x-\tfrac12)(x-1),\quad
      \ell_1(x)=-4x(x-1),\quad
      \ell_2(x)=2x(x-\tfrac12).
    \]
    Donc
    \[
      P_2(x)=g(t_n)\ell_0(x)+g\!\bigl(t_n+\tfrac h2\bigr)\ell_1(x)+g(t_{n+1})\ell_2(x).
    \]
    En intégrant sur $x\in[0,1]$ :
    \[
      \int_0^1 \ell_0(x)\,dx=\frac16,\quad
      \int_0^1 \ell_1(x)\,dx=\frac{4}{6},\quad
      \int_0^1 \ell_2(x)\,dx=\frac16,
    \]
    d'où
    \[
      \int_{t_n}^{t_{n+1}} g(t)\,dt
      \approx
      \frac h6\Bigl(g(t_n)+4g\!\bigl(t_n+\tfrac h2\bigr)+g(t_{n+1})\Bigr).
    \]
    

    Pour une version entièrement explicite et intuitive, on évalue la pente
    au début, puis deux fois au milieu (avec correction), puis à la fin :
    \[
      k_1=f(t_n,y_n),\qquad
      k_2=f\!\Bigl(t_n+\frac h2,\,y_n+\frac h2k_1\Bigr),
    \]
    \[
      k_3=f\!\Bigl(t_n+\frac h2,\,y_n+\frac h2k_2\Bigr),\qquad
      k_4=f(t_n+h,\,y_n+hk_3).
    \]
    Puis on combine ces pentes avec les poids $1,2,2,1$ :
    \[
      y_{n+1}=y_n+\frac h6\,(k_1+2k_2+2k_3+k_4).
    \]
    On appelle ça la méthode Runge Kutta (RK4).
  \end{solution}
}{}


\subsection*{5) Application numérique (RK4)}
\begin{enumerate}[itemsep=0.5em]
\item Avec $h=0{,}5$ et $y_0=1$, calculer les deux premiers termes $y_1$ et $y_2$
obtenus par la méthode de Runge--Kutta d'ordre 4.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Ici $f(t,y)=-2y$.

    \medskip
    \textbf{Premier pas} ($t_0=0$, $y_0=1$) :
    \[
      k_1=-2,\quad
      k_2=-2\Bigl(1+\frac{0{,}5}{2}k_1\Bigr)=-1,\quad
      k_3=-2\Bigl(1+\frac{0{,}5}{2}k_2\Bigr)=-1{,}5,
    \]
    \[
      k_4=-2\bigl(1+0{,}5k_3\bigr)=-0{,}5.
    \]
    Donc
    \[
      y_1
      =1+\frac{0{,}5}{6}\,(k_1+2k_2+2k_3+k_4)
      =1+\frac{0{,}5}{6}(-7{,}5)
      =0{,}375.
    \]

    \medskip
    \textbf{Deuxième pas} ($t_1=0{,}5$, $y_1=0{,}375$) :
    \[
      k_1=-0{,}75,\quad
      k_2=-2\Bigl(0{,}375+\frac{0{,}5}{2}k_1\Bigr)=-0{,}375,
    \]
    \[
      k_3=-2\Bigl(0{,}375+\frac{0{,}5}{2}k_2\Bigr)=-0{,}5625,\quad
      k_4=-2\bigl(0{,}375+0{,}5k_3\bigr)=-0{,}1875.
    \]
    Donc
    \[
      y_2
      =0{,}375+\frac{0{,}5}{6}\,(k_1+2k_2+2k_3+k_4)
      =0{,}375+\frac{0{,}5}{6}(-2{,}8125)
      \approx 0{,}1406.
    \]
  \end{solution}
}{}

\item Reprendre les résultats précédents et comparer, pour $h=0{,}5$, les valeurs obtenues avec Euler explicite,
Euler implicite, Crank--Nicolson et RK4 aux instants $t=0{,}5$ et $t=1$.
Quelle méthode est la plus précise ici ?

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    On compare avec les valeurs exactes
    \[
      y(0{,}5)=e^{-1}\approx 0{,}3679,\qquad y(1)=e^{-2}\approx 0{,}1353.
    \]

    \begin{tabular}{lcccc}
      Méthode & $y_1$ & $|y_1-y(0{,}5)|$ & $y_2$ & $|y_2-y(1)|$ \\
      \hline
      Euler explicite & $0$ & $0{,}3679$ & $0$ & $0{,}1353$ \\
      Euler implicite & $0{,}5$ & $0{,}1321$ & $0{,}25$ & $0{,}1147$ \\
      Crank--Nicolson & $1/3 \approx 0{,}3333$ & $0{,}0346$ & $1/9 \approx 0{,}1111$ & $0{,}0242$ \\
      RK4 & $0{,}3750$ & $0{,}0071$ & $0{,}1406$ & $0{,}0053$
    \end{tabular}

    \medskip
    Ici, RK4 est la plus précise parmi les quatre méthodes.
  \end{solution}
}{}
\end{enumerate}

\section*{Exercice 2 --- Réutiliser les schémas sur une nouvelle EDO}

On considère maintenant
\[
  y'(t)=1-y(t), \qquad y(0)=0.
\]
On admet que la solution exacte est
\[
  y(t)=1-e^{-t}.
\]

\begin{enumerate}[itemsep=0.5em]
\item Écrire les formules de récurrence de Euler explicite, Euler implicite et Crank--Nicolson pour cette EDO.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    \[
      \text{Euler explicite :}\quad
      y_{n+1}=y_n+h(1-y_n)=(1-h)y_n+h.
    \]
    \[
      \text{Euler implicite :}\quad
      y_{n+1}=y_n+h(1-y_{n+1})
      \;\Rightarrow\;
      y_{n+1}=\frac{y_n+h}{1+h}.
    \]
    \[
      \text{Crank--Nicolson :}\quad
      y_{n+1}=y_n+\frac h2\big[(1-y_n)+(1-y_{n+1})\big]
      \;\Rightarrow\;
      y_{n+1}=\frac{(1-\frac h2)y_n+h}{1+\frac h2}.
    \]
  \end{solution}
}{}

\item Avec $h=0{,}5$, calculer $y_1$ et $y_2$ pour les trois méthodes.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    Valeurs exactes de référence : $y(0{,}5)=1-e^{-0{,}5}\approx 0{,}3935$ et $y(1)=1-e^{-1}\approx 0{,}6321$.

    \begin{itemize}
      \item Euler explicite : $y_1=0{,}5$, $y_2=0{,}75$.
      \item Euler implicite : $y_1=\frac13\approx0{,}3333$, $y_2=\frac59\approx0{,}5556$.
      \item Crank--Nicolson : $y_1=0{,}4$, $y_2=0{,}64$.
    \end{itemize}
  \end{solution}
}{}

\item Comparer les erreurs absolues à $t=0{,}5$ et $t=1$. Quelle méthode est la plus précise ici ?

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    \begin{tabular}{lcc}
      Méthode & $|y_1-y(0{,}5)|$ & $|y_2-y(1)|$ \\
      \hline
      Euler explicite & $|0{,}5-0{,}3935|\approx 0{,}1065$ & $|0{,}75-0{,}6321|\approx 0{,}1179$ \\
      Euler implicite & $|0{,}3333-0{,}3935|\approx 0{,}0602$ & $|0{,}5556-0{,}6321|\approx 0{,}0765$ \\
      Crank--Nicolson & $|0{,}4-0{,}3935|\approx 0{,}0065$ & $|0{,}64-0{,}6321|\approx 0{,}0079$
    \end{tabular}

    \medskip
    Crank--Nicolson est nettement plus précis sur cet exemple.
  \end{solution}
}{}

\item Même question avec $h=0{,}25$. Vérifier que réduire le pas améliore toutes les méthodes.

\ifthenelse{\boolean{showSolutions}}{
  \begin{solution}
    On observe une baisse des erreurs pour les trois schémas.
    Le gain est plus marqué pour Crank--Nicolson (ordre 2) que pour Euler (ordre 1).
  \end{solution}
}{}
\end{enumerate}

\section*{Exercice 3 --- Intégration numérique. Une autre méthode de quadrature}
Soit $f \in \mathbb{C}([-1,1])$, on considère la formule de quadrature suivante :
$$
M(f)=\alpha f\left(-\frac{1}{2}\right)+\beta f(0)+\gamma f\left(\frac{1}{2}\right)
$$
C'est-à-dire que $M$ est une valeur approchée de l'intégrale de $f$ sur $[-1,1]$.

\begin{enumerate}
    \item Déterminer $\alpha, \beta$ et $\gamma$ de telle sorte que $M(f)=\int_{-1}^1 f(x) d x$, si $f$ est un polynôme de degré inférieur ou égal à 2.
    \ifthenelse{\boolean{showSolutions}}{
    \begin{solution}
        La formule doit fonctionner si $f$ est une constante $a$, si $f$ est de type $b x$ et si $f$ est de type $c x^2$.

        Si $f$ est une constante, il faut : 
        \[
        \alpha a + \beta a + \gamma a = 2a
        \]
        C'est-à-dire : 
        \[
        \alpha + \beta + \gamma = 2
        \]

        Si $f$ est de type $b x$, il faut : 
        \[
        -\alpha \frac{b}{2} + \beta 0 + \gamma \frac{b}{2} = 0
        \]
        C'est-à-dire : 
        \[
        -\alpha \frac{b}{2} + \gamma \frac{b}{2} = 0
        \]
        C'est-à-dire : 
        \[
        -\alpha + \gamma = 0
        \]

        Si $f$ est de type $c x^2$, il faut : 
        \[
        \alpha \frac{c}{4} + \beta 0 + \gamma \frac{c}{4} = \frac{2c}{3}
        \]
        C'est-à-dire : 
        \[
        \alpha \frac{c}{4} + \gamma \frac{c}{4} = \frac{2c}{3}
        \]
        C'est-à-dire : 
        \[
        \alpha + \gamma = \frac{8}{3}
        \]

        On résout le système : 
        \[
        \begin{cases}
            \alpha + \beta + \gamma = 2 \\
            -\alpha + \gamma = 0 \\
            \alpha + \gamma = \frac{8}{3}
        \end{cases}
        \]
        On trouve : 
        \[
        \alpha = \frac{4}{3}, \quad \beta = -\frac{2}{3}, \quad \gamma = \frac{4}{3}
        \]
        
    \end{solution}
    }{}
    \item Cette égalité est-elle aussi vraie pour un polynôme de degré 3 ou 4 ?
    \ifthenelse{\boolean{showSolutions}}{
    \begin{solution}
        Essayons avec $f(x) = x^3$ : 
        \[
        \int_{-1}^1 x^3 dx = 0
        \]
        \[
        \frac{4}{3} \left(-\frac{1}{2}\right)^3 + \left(-\frac{2}{3}\right) 0 + \frac{4}{3} \left(\frac{1}{2}\right)^3 = 0
        \]
        La formule fonctionne donc pour tous les polynômes de degré inférieur ou égal à 3.

        Essayons avec $f(x) = x^4$ : 
        \[
        \int_{-1}^1 x^4 dx = \frac{2}{5}
        \]
        \[
        \frac{4}{3} \left(-\frac{1}{2}\right)^4 + \left(-\frac{2}{3}\right) 0 + \frac{4}{3} \left(\frac{1}{2}\right)^4 = \frac{1}{6}
        \]
        La formule ne fonctionne pas pour les polynômes de degré 4.
    \end{solution}
    }{}
    \item Calculer l'intégrale $\int_{-1}^1 \frac{1}{1+x^2} d x$.
    \ifthenelse{\boolean{showSolutions}}{
    \begin{solution}
        On connait une primitive de cette fonction : 
        \[
        \int_{-1}^1 \frac{1}{1+x^2} dx = \left[ \arctan(x) \right]_{-1}^1 = \arctan(1) - \arctan(-1) = \frac{\pi}{4} - \left(-\frac{\pi}{4}\right) = \frac{\pi}{2}
        \]
    \end{solution}
    }{}
