\section*{1. Le problème : peut-on deviner une élection avec $500$ sondés\,?}
% ====================================================================

\textbf{Cadre.} Une élection oppose deux candidats A et B dans une
population de plusieurs millions d'électeurs. Chaque électeur a un
vote déterministe (A ou B). La proportion \emph{réelle} de votants
pour B est un nombre $p\in[0,1]$ \textbf{inconnu}, et c'est
précisément ce qu'on cherche à estimer \textbf{avant} le jour du
scrutin.

\medskip

On interroge $n$ électeurs, \emph{tirés au hasard} dans la
population (en prenant soin de \og panacher \fg{} : région, âge,
profession\dots, pour que l'échantillon soit représentatif). À
chaque sondé $i$ on associe
\[
  X_i=\begin{cases}1 & \text{si l'électeur } i \text{ vote B},\\
  0 & \text{si l'électeur } i \text{ vote A}.\end{cases}
\]

\textbf{Question 1.} Quelle loi suit $X_i$\,? Donner $\E(X_i)$ et
$\Var(X_i)$ en fonction de $p$.

\atrou[2.5cm]{%
$X_i\sim\mathcal{B}(p)$ (Bernoulli de paramètre $p$). Donc
\[
  \E(X_i)=p\qquad\text{et}\qquad \Var(X_i)=p(1-p).
\]
}

\textbf{Question 2.} Ces variables aléatoires $X_1,\dots,X_n$
sont-elles indépendantes\,? Suivent-elles toutes la même loi\,?

\atrou[2.5cm]{%
Si les électeurs sont tirés au hasard dans une grande population
(avec remise, ou sans remise avec une population $\gg n$), oui :
les $X_i$ sont \textbf{indépendants} et suivent la \textbf{même}
loi $\mathcal{B}(p)$. C'est le rôle du panachage que de justifier
l'indépendance.
}

On dit que les $X_i$ sont \textbf{i.i.d.} (indépendantes,
identiquement distribuées). On note $m=\E(X_1)=p$ et
$\sigma^2=\Var(X_1)=p(1-p)$.

\medskip

On introduit la \textbf{moyenne empirique} :
\[
  \boxed{\,\Xbar = \dfrac{X_1 + X_2 + \cdots + X_n}{n}.\,}
\]

\textbf{Question 3.} À quoi correspond, en mots, la somme
$X_1+\cdots+X_n$\,? Et $\Xbar$\,?

\atrou[2.2cm]{%
$X_1+\cdots+X_n$ est le \textbf{nombre} de personnes sondées
qui votent B.
$\Xbar$ est la \textbf{proportion} (ou \emph{fréquence observée})
de sondés votant pour B dans l'échantillon.
}

\begin{center}
\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1.5pt, innertopmargin=10pt, innerbottommargin=10pt]
\centering\large\itshape
Si je sonde \textbf{$500$} personnes soigneusement panachées et que
j'observe $52\,\%$ d'intentions de vote pour B, puis-je en déduire
que B va gagner\,?\newline
Et si j'en sondais seulement $100$\,? Ou $10\,000$\,?
\end{mdframed}
\end{center}

\textbf{Les deux questions centrales du cours.}
\begin{enumerate}
  \item Vers quoi tend $\Xbar$ quand $n \to +\infty$\,? \hfill
    (\textit{loi des grands nombres})
  \item Pour $n$ \emph{fini}, à quelle distance de $p$ est $\Xbar$\,?
    À $\pm 1\,\%$ près\,? À $\pm 5\,\%$ près\,? \hfill
    (\textit{théorème central limite})
\end{enumerate}

Ces deux questions se posent en fait dans tous les problèmes de
mesures répétées (contrôle qualité, métrologie, physique
expérimentale, machine learning\dots). On les traitera ici sur le
\emph{fil rouge} des sondages.

% ====================================================================
\section*{2. Loi des grands nombres}
% ====================================================================

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Théorème (Loi des grands nombres).}
Soient $X_1, X_2, \dots$ des variables aléatoires \textbf{i.i.d.}
admettant une espérance $m=\E(X_1)$. Alors
\[
  \boxed{\,\Xbar = \dfrac{X_1+\cdots+X_n}{n}\;\xrightarrow[n\to+\infty]{}\; m.\,}
\]
\end{mdframed}

\textbf{Interprétation.}
Si on \emph{moyenne} un grand nombre de réalisations indépendantes
de la même expérience, on se rapproche de l'espérance théorique.

\medskip

\textbf{Question 1.} Appliquer la LGN au cas du sondage : vers
quoi tend $\Xbar$\,? Interpréter cette limite en une phrase.

\atrou[3cm]{%
Ici $m=\E(X_i)=p$. La LGN dit :
$\Xbar=\text{fréquence observée de B}\;\xrightarrow[n\to+\infty]{}\; p
=\text{proportion réelle de B}.$

\textbf{C'est précisément ce qui légitime la pratique des sondages} :
plus on interroge de monde, plus la fréquence observée est proche
de la proportion réelle.
}

\textbf{Question 2.} On lance une pièce \textbf{équilibrée} un grand
nombre de fois. Que dit la LGN sur la fréquence de Pile\,? Et si le
lancé concerne un dé à $6$ faces équilibré, que dit-elle sur la
moyenne des résultats\,?

\atrou[3cm]{%
Pièce équilibrée : $X_i\in\{0,1\}$ avec $\E(X_i)=1/2$, donc la
fréquence de Pile tend vers $1/2$.

Dé à $6$ faces équilibré : $\E(X_1)=(1+2+3+4+5+6)/6=3{,}5$. La
moyenne des résultats tend vers $3{,}5$.
}

\medskip

\textbf{Question 3 --- Ce que la LGN ne dit pas.}
La LGN garantit la convergence \emph{à l'infini}. Mais pour
$n=500$, que dit-elle \emph{quantitativement} sur la distance
$|\Xbar-p|$\,?

\atrou[2cm]{%
\textbf{Rien.} La LGN est un théorème de pure convergence : elle ne
donne aucune vitesse, aucune marge d'erreur pour $n$ fini. C'est
tout le problème --- et c'est le théorème central limite qui y
répondra (§~4).
}

% ====================================================================
\section*{3. La loi normale}
% ====================================================================

On introduit ici la \textbf{loi normale} (ou \textbf{gaussienne}),
qui apparaîtra naturellement dans le TCL (§~4).

\subsection*{3.1. Densité et dessin}

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Définition.} Pour $\mu\in\R$ et $\sigma>0$, on dit que
$X\sim\mathcal{N}(\mu,\sigma^2)$ lorsque $X$ admet pour densité
\[
  \boxed{\,f(x)=\dfrac{1}{\sigma\sqrt{2\pi}}\,
  \exp\!\left(-\dfrac{(x-\mu)^2}{2\sigma^2}\right),\quad x\in\R.\,}
\]
Un calcul d'intégrale (admis) donne alors
\[
  \boxed{\E(X)=\mu,}\qquad \boxed{\Var(X)=\sigma^2,}
  \qquad \boxed{\sigma(X)=\sigma.}
\]
\end{mdframed}

\textbf{Question 1.} Montrer que $f$ est bien une fonction positive
symétrique autour de $\mu$, c'est-à-dire $f(\mu+h)=f(\mu-h)$ pour
tout $h\in\R$.

\atrou[2.5cm]{%
Positivité : $f(x)$ est le produit d'une constante positive par
une exponentielle, donc $f(x)>0$ pour tout $x$.\\
Symétrie : $(\mu+h-\mu)^2=h^2=(\mu-h-\mu)^2$, donc
$f(\mu+h)=f(\mu-h)$. La cloche est donc symétrique autour
de $x=\mu$ (moyenne = médiane = mode).
}

\textbf{Question 2.} Pour quelle valeur de $x$ la densité
$f$ atteint-elle son maximum\,? Que vaut alors $f(x)$\,?

\atrou[2.5cm]{%
L'exponentielle est maximale lorsque l'exposant est nul, i.e.\
en $x=\mu$. La valeur maximale est
\[
  f(\mu)=\dfrac{1}{\sigma\sqrt{2\pi}}.
\]
Plus $\sigma$ est grand, plus cette valeur est petite : la cloche
s'\textbf{aplatit} quand $\sigma$ augmente.
}

\begin{center}
\begin{tikzpicture}[scale=0.9]
  \draw[thick,->] (-3.8,0) -- (3.8,0) node[right] {\small $x$};
  \draw[thick,->] (0,-0.2) -- (0,2.6) node[above] {\small $f(x)$};
  \draw[thick, Theme, domain=-3.4:3.4, smooth, samples=100]
    plot (\x, {2.2*exp(-\x*\x/2)});
  \draw[dashed] (1,0) -- (1,{2.2*exp(-0.5)});
  \draw[dashed] (-1,0) -- (-1,{2.2*exp(-0.5)});
  \draw[dashed] (0,2.2) -- (0,0);
  \node at (0,-0.35) {\small $\mu$};
  \node at (1,-0.35) {\small $\mu+\sigma$};
  \node at (-1,-0.35) {\small $\mu-\sigma$};
\end{tikzpicture}
\end{center}

\textbf{Question 3.} Soit $X\sim\mathcal{N}(10,\,4)$. Donner
$\E(X)$, $\Var(X)$ et $\sigma(X)$.

\atrou[2cm]{%
Attention : le $2^{\text{e}}$ paramètre est la \emph{variance}.
Donc
$\E(X)=10$, $\Var(X)=4$, $\sigma(X)=\sqrt{4}=2$.
}

\subsection*{3.2. Cas particulier : la loi normale centrée réduite}

Le cas $\mu=0$ et $\sigma=1$ est fondamental.

\textbf{Question 4.} Écrire la densité de $\mathcal{N}(0,1)$ et
donner $\E(Z)$, $\Var(Z)$.

\atrou[2.5cm]{%
\[
  f(z)=\dfrac{1}{\sqrt{2\pi}}\,e^{-z^2/2},\qquad z\in\R.
\]
$\E(Z)=0$ et $\Var(Z)=1$ (d'où le nom \og centrée réduite \fg).
}

\subsection*{3.3. Un problème nouveau : la fonction de répartition}

Pour toute loi continue, la fonction de répartition est
\[
  F(x) = P(X\le x) = \int_{-\infty}^{x} f(t)\,dt.
\]
Pour la loi normale, on bute sur un obstacle inédit : la
\textbf{primitive de $e^{-z^2/2}$ n'a pas d'expression
élémentaire}. On ne peut donc pas donner $F$ par une formule
fermée.

\medskip

\textbf{La solution : tabuler la fonction.}
On donne à la répartition de $\mathcal{N}(0,1)$ un nom et un
symbole propres :
\[
  \boxed{\,\Phi(z)=P(Z\le z)
  =\int_{-\infty}^{z}\dfrac{1}{\sqrt{2\pi}}\,e^{-t^2/2}\,dt,\qquad Z\sim\mathcal{N}(0,1).\,}
\]
C'est une \textbf{fonction classique}, au même titre que $\exp$,
$\log$ ou $\sin$. Elle n'admet pas de formule fermée mais ses
valeurs sont \textbf{tabulées une fois pour toutes} (§~6) ou
calculées par des algorithmes dans les calculatrices et
les logiciels (en Python : \texttt{scipy.stats.norm.cdf}).

\medskip

\textbf{Question 5.} Sans table, calculer $\Phi(0)$. Justifier.

\atrou[2.5cm]{%
Par symétrie de la cloche autour de $0$ :
$P(Z\le 0)=P(Z\ge 0)$, et ces deux probabilités somment à $1$.
Donc $\Phi(0)=P(Z\le 0)=1/2$. Autant de chances d'être positive
que négative.
}

\section*{4. Théorème central limite}
% ====================================================================

Le TCL précise \emph{la vitesse} et \emph{la loi} des fluctuations
de $\Xbar$ autour de $m$. C'est le théorème qui répond à la question
\og à quelle distance de $p$ est $\Xbar$ pour $n=500$\,? \fg.

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Théorème central limite (TCL).}
Soient $X_1, X_2, \dots$ des variables aléatoires \textbf{i.i.d.}
admettant une espérance $m$ et une variance $\sigma^2$ finies. Alors
la variable aléatoire
\[
  \boxed{\,Z_n=\dfrac{\Xbar - m}{\sigma/\sqrt{n}}
  =\dfrac{\sqrt{n}}{\sigma}\bigl(\Xbar - m\bigr)\,}
\]
converge \emph{en loi} vers la loi normale centrée réduite
$\mathcal{N}(0,1)$. Autrement dit, pour tous réels $a\le b$,
\[
  P(a\le Z_n\le b)\;\xrightarrow[n\to+\infty]{}\;
  \Phi(b)-\Phi(a).
\]
\end{mdframed}

\textbf{Forme ramassée (à retenir).}
Pour $n$ grand, l'écart $\Xbar-m$ est approximativement gaussien,
d'écart-type $\sigma/\sqrt{n}$ :
\[
  \boxed{\,\Xbar\;\approx\; m + \dfrac{\sigma}{\sqrt{n}}\,Z\qquad
  \text{avec}\quad Z\sim\mathcal{N}(0,1).\,}
\]

\medskip

\textbf{Question 1 --- Vitesse en $1/\sqrt{n}$.}
Par quel facteur l'écart-type de $\Xbar$ est-il divisé si on
multiplie $n$ par $100$\,? Et pour diviser l'écart-type par $10$,
par quel facteur faut-il multiplier $n$\,?

\atrou[3cm]{%
L'écart-type de $\Xbar$ est $\sigma/\sqrt{n}$. Si $n\leftarrow 100\,n$,
alors $\sqrt{n}\leftarrow 10\sqrt{n}$ : l'écart-type est divisé par
$\mathbf{10}$.

Pour diviser l'écart-type par $10$, il faut donc multiplier $n$
par $\mathbf{100}$. C'est la \textbf{vitesse en $\sqrt{n}$}.
}

\textbf{Question 2 --- Application au sondage.}
Dans le cas du sondage, on a $m=p$ et $\sigma^2=p(1-p)$. Écrire
l'approximation normale de $\Xbar$ donnée par le TCL.

\atrou[2.5cm]{%
\[
  \Xbar\;\approx\; p + \sqrt{\dfrac{p(1-p)}{n}}\,Z,\qquad
  Z\sim\mathcal{N}(0,1).
\]
L'écart-type de $\Xbar$ autour de $p$ est $\sqrt{p(1-p)/n}$. Cette
formule sera exploitée en détail au § 7.
}

\medskip

\textbf{Question 3 --- Universalité.}
Dans l'énoncé du TCL, où intervient la loi précise des $X_i$ dans
la limite obtenue\,?

\atrou[2.5cm]{%
\textbf{Nulle part !} La loi limite est toujours $\mathcal{N}(0,1)$,
quelle que soit la loi des $X_i$ (à condition que $\E(X_1)$ et
$\Var(X_1)$ existent). C'est la \textbf{propriété d'universalité}
du TCL : c'est \emph{toujours} la gaussienne qui apparaît.
}

\medskip

\textbf{Expérience de pensée --- planche de Galton.}
On lâche une bille qui, à chaque rangée de clous ($n$ rangées),
dévie de $+1$ ou $-1$ avec probabilité $1/2$. La position finale
$S_n=\varepsilon_1+\cdots+\varepsilon_n$ est comptée en bas.

\section*{5. Standardisation : tout se ramène à $\mathcal{N}(0,1)$}
% ====================================================================

La table de $\Phi$ ne donne que la loi $\mathcal{N}(0,1)$. Comment
calcule-t-on $P(a\le X\le b)$ pour $X\sim\mathcal{N}(\mu,\sigma^2)$
quelconque\,?

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Proposition (standardisation).}
Si $X\sim\mathcal{N}(\mu,\sigma^2)$, alors
\[
  \boxed{\,Z=\dfrac{X-\mu}{\sigma}\sim\mathcal{N}(0,1).\,}
\]
Réciproquement, si $Z\sim\mathcal{N}(0,1)$, alors
$X=\mu+\sigma Z\sim\mathcal{N}(\mu,\sigma^2)$.
\end{mdframed}

\textbf{Idée.}
Centrer ($X-\mu$ retire la moyenne), puis réduire (diviser par
$\sigma$ pour avoir un écart-type de $1$). $Z$ s'interprète comme
\og le nombre d'écarts-types au-dessus de la moyenne \fg.

\medskip

\textbf{Question 1.} Soit $X\sim\mathcal{N}(10,\,25)$. Quelle est
la loi de $Z=(X-10)/5$\,? Et celle de $Y=(X-10)/25$\,?

\atrou[3cm]{%
Ici $\mu=10$ et $\sigma=\sqrt{25}=5$ (attention : variance $=25$,
écart-type $=5$). Donc
\[
  Z=\dfrac{X-10}{5}\sim\mathcal{N}(0,1).
\]
La variable $Y=(X-10)/25=Z/5$ n'est \emph{pas} standard :
$\Var(Y)=\Var(Z)/25=1/25$, donc $Y\sim\mathcal{N}(0,\,1/25)$.
Bien diviser par l'\textbf{écart-type}, pas par la variance.
}

\medskip

\textbf{Question 2.} Soit $X\sim\mathcal{N}(10,\,25)$. Exprimer
$P(X\le 15)$ et $P(X\ge 20)$ en fonction de $\Phi$.

\atrou[3.5cm]{%
$Z=(X-10)/5\sim\mathcal{N}(0,1)$, donc
\[
  P(X\le 15)=P\!\left(Z\le \tfrac{15-10}{5}\right)=\Phi(1).
\]
\[
  P(X\ge 20)=P\!\left(Z\ge \tfrac{20-10}{5}\right)=P(Z\ge 2)=1-\Phi(2).
\]
}

\textbf{Principe général.}
Toute probabilité sur $X\sim\mathcal{N}(\mu,\sigma^2)$ se ramène
à une probabilité sur $Z$ :
\[
  P(X\le x) = P\!\left(Z\le \dfrac{x-\mu}{\sigma}\right)
  = \Phi\!\left(\dfrac{x-\mu}{\sigma}\right).
\]

\textbf{Question 1 --- Échauffement.}
Lire dans la table : $\Phi(1)$, $\Phi(1{,}96)$, $\Phi(2)$ et
$\Phi(2{,}58)$.

\atrou[2cm]{%
$\Phi(1)=0{,}8413$,\quad $\Phi(1{,}96)=0{,}9750$,\quad
$\Phi(2)=0{,}9772$,\quad $\Phi(2{,}58)\approx 0{,}9951$.
}

\textbf{Question 2 --- Cas $z<0$.}
La table ne donne que $z\ge 0$. Comment calculer $\Phi(-a)$ pour
$a>0$\,? (On s'aidera d'un dessin de la cloche.)

\atrou[3cm]{%
Par \textbf{symétrie} de la cloche autour de $0$ : l'aire à gauche
de $-a$ est égale à l'aire à droite de $+a$. Donc
$\Phi(-a)=P(Z\le -a)=P(Z\ge a)=1-\Phi(a).$
}

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Les quatre réflexes.}
Pour $Z\sim\mathcal{N}(0,1)$ et tous réels $a\le b$ :
\begin{enumerate}
  \item $P(Z\le a)=\Phi(a)$ (lecture directe si $a\ge 0$).
  \item \textbf{Symétrie} : $\boxed{\Phi(-a)=1-\Phi(a).}$
  \item $\boxed{P(a\le Z\le b)=\Phi(b)-\Phi(a).}$
  \item $\boxed{P(|Z|\le a)=2\Phi(a)-1 \;(a\ge 0).}$
\end{enumerate}
\end{mdframed}

\medskip

\textbf{Question 3 --- Problème d'application.}
Soit $Z\sim\mathcal{N}(0,1)$. Calculer :
\begin{enumerate}
  \item $P(Z\le -0{,}58)$.
  \item $P(Z\ge 1{,}96)$.
  \item $P(-1\le Z\le 1)$.
  \item $P(|Z|\le 1{,}96)$ et $P(|Z|\le 2{,}58)$.
\end{enumerate}

\atrou[7.5cm]{%
\begin{enumerate}
  \item $P(Z\le -0{,}58)=\Phi(-0{,}58)=1-\Phi(0{,}58)
    =1-0{,}7190=0{,}2810.$
  \item $P(Z\ge 1{,}96)=1-\Phi(1{,}96)=1-0{,}9750=0{,}0250.$
  \item $P(-1\le Z\le 1)=2\Phi(1)-1=2\cdot 0{,}8413-1=0{,}6826.$
    Environ $68\,\%$ des valeurs sont dans $[\mu-\sigma,\mu+\sigma]$.
  \item $P(|Z|\le 1{,}96)=2\Phi(1{,}96)-1=0{,}95.$
    \emph{Règle des $95\,\%$.} \\
    $P(|Z|\le 2{,}58)=2\Phi(2{,}58)-1\approx 0{,}99.$
    Règle des $99\,\%$.
\end{enumerate}
}

\medskip

\textbf{Règle 68 / 95 / 99.}
Pour $X\sim\mathcal{N}(\mu,\sigma^2)$ :
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|}
  \hline
  Intervalle autour de $\mu$ & Probabilité \\
  \hline
  $[\mu-\sigma,\mu+\sigma]$ & $\approx 68\,\%$ \\
  \hline
  $[\mu-1{,}96\sigma,\mu+1{,}96\sigma]$ & $=95\,\%$ \\
  \hline
  $[\mu-2{,}58\sigma,\mu+2{,}58\sigma]$ & $=99\,\%$ \\
  \hline
  $[\mu-3\sigma,\mu+3\sigma]$ & $\approx 99{,}7\,\%$ \\
  \hline
\end{tabular}
\end{center}

\section*{7. Retour aux sondages : combien de personnes sonder\,?}
% ====================================================================

On reprend le fil rouge du § 1 : $X_i\in\{0,1\}$ i.i.d.\
de Bernoulli $\mathcal{B}(p)$, donc
$m=\E(X_i)=p$ et $\sigma^2=\Var(X_i)=p(1-p)$.
La \textbf{fréquence observée} est $\Xbar$, qui estime $p$.

\medskip

\textbf{Question 1 --- Appliquer le TCL.}
Écrire l'approximation normale de $\Xbar$ pour $n$ grand dans ce
contexte.

\atrou[2.5cm]{%
\[
  \Xbar \;\approx\; p + \sqrt{\dfrac{p(1-p)}{n}}\,Z,
  \qquad Z\sim\mathcal{N}(0,1).
\]
L'écart $\Xbar-p$ est gaussien, d'écart-type
$\sqrt{p(1-p)/n}$.
}

\textbf{Question 2 --- Intervalle de confiance à $95\,\%$.}
En utilisant la règle des $95\,\%$, montrer que, avec une
probabilité d'environ $95\,\%$,
\[
  \bigl|\Xbar-p\bigr|\;\le\; 1{,}96\,\sqrt{\dfrac{p(1-p)}{n}}.
\]

\atrou[3cm]{%
Par la règle : $P(|Z|\le 1{,}96)=0{,}95$. Or
$Z=(\Xbar-p)/\sqrt{p(1-p)/n}$, donc
\[
  P\!\left(|\Xbar-p|\le 1{,}96\sqrt{\dfrac{p(1-p)}{n}}\right)\approx 0{,}95.
\]
La demi-longueur $1{,}96\sqrt{p(1-p)/n}$ est la \textbf{marge
d'erreur} du sondage à $95\,\%$.
}

\medskip

\textbf{Question 3 --- Majoration universelle.}
Étudier la fonction $p\mapsto p(1-p)$ sur $[0,1]$. En quel point
est-elle maximale\,? Que vaut alors ce maximum\,? En déduire que
$\sqrt{p(1-p)/n}\le 1/(2\sqrt{n})$ pour tout $p\in[0,1]$.

\atrou[3.5cm]{%
$g(p)=p(1-p)=p-p^2$, $g'(p)=1-2p$ qui s'annule en $p=1/2$ ;
$g(1/2)=1/4$ est un maximum. Donc pour tout $p$,
$p(1-p)\le 1/4$ et
\[
  \sqrt{\dfrac{p(1-p)}{n}}\le \sqrt{\dfrac{1}{4n}}=\dfrac{1}{2\sqrt{n}}.
\]
}

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1pt, innertopmargin=8pt, innerbottommargin=8pt, nobreak=true]
\textbf{Marge d'erreur universelle à $95\,\%$.}
Quelle que soit la valeur vraie de $p$, avec probabilité
$\approx 95\,\%$,
\[
  \boxed{\,\bigl|\Xbar-p\bigr|\;\le\;\dfrac{1{,}96}{2\sqrt{n}}\;\approx\;\dfrac{1}{\sqrt{n}}.\,}
\]
C'est la formule souvent citée dans la presse : \og marge d'erreur
de $\pm 1/\sqrt{n}$ \fg.
\end{mdframed}

\medskip

\textbf{Problème 1 --- Taille d'un sondage.}
Un institut veut estimer $p$ à $\pm 3\,\%$ près, avec un niveau de
confiance de $95\,\%$. Combien de personnes faut-il sonder\,?

\atrou[4cm]{%
On impose $\dfrac{1}{\sqrt{n}}\le 0{,}03$, soit
\[
  \sqrt{n}\ge \dfrac{1}{0{,}03}\approx 33{,}3
  \;\Longrightarrow\;
  n\ge 33{,}3^2\approx 1\,111.
\]
On sonde donc environ $1\,100$ personnes --- c'est l'ordre de
grandeur standard des instituts.
}

\medskip

\textbf{Problème 2 --- Marge d'erreur en fonction de $n$.}
Compléter le tableau suivant ($\varepsilon=1/\sqrt{n}$) :

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|}
  \hline
  $n$ & marge d'erreur $\varepsilon$ \\
  \hline
  $100$ & \val{$\pm 10\,\%$} \\
  \hline
  $500$ & \val{$\pm 4{,}5\,\%$} \\
  \hline
  $1\,000$ & \val{$\pm 3{,}2\,\%$} \\
  \hline
  $10\,000$ & \val{$\pm 1\,\%$} \\
  \hline
  $1\,000\,000$ & \val{$\pm 0{,}1\,\%$} \\
  \hline
\end{tabular}
\end{center}

\textbf{À retenir.} Pour diviser la marge d'erreur par $10$, il
faut multiplier la taille du sondage par $100$. C'est pour cela
qu'aucun institut ne va jamais sonder $1$ million de personnes :
le coût est prohibitif pour une amélioration marginale.

\medskip

\textbf{Problème 3 --- Retour à la question du § 1.}
Un sondage sur $n=500$ personnes donne $52\,\%$ d'intentions de
vote pour B.
\begin{enumerate}
  \item Quelle est la marge d'erreur à $95\,\%$\,?
  \item Peut-on \og prédire \fg{} la victoire de B\,?
  \item Même question pour $n=1\,000$ personnes avec $52\,\%$.
  \item Même question pour $n=10\,000$ personnes avec $52\,\%$.
\end{enumerate}

\atrou[7cm]{%
\begin{enumerate}
  \item $1/\sqrt{500}\approx 0{,}0447$, soit $\pm 4{,}5\,\%$.
    L'intervalle de confiance à $95\,\%$ est
    $[47{,}5\,\%\,;\,56{,}5\,\%]$.
  \item $50\,\%$ est dans cet intervalle : \textbf{non}, on ne peut
    pas conclure à la victoire de B avec $95\,\%$ de confiance.
    Le sondage est \og dans la marge d'erreur \fg.
  \item $1/\sqrt{1000}\approx 0{,}032$ : IC
    $[48{,}8\,\%\,;\,55{,}2\,\%]$. $50\,\%$ toujours dedans :
    on ne conclut toujours pas.
  \item $1/\sqrt{10\,000}=0{,}01$ : IC
    $[51\,\%\,;\,53\,\%]$. $50\,\%$ \emph{hors} de l'intervalle :
    on peut conclure à la victoire de B avec $95\,\%$ de confiance.
\end{enumerate}
\textbf{Moralité} : un écart de $2$ points sur $500$ ou $1\,000$
sondés ne signifie \textbf{rien}. Il faut soit un écart plus
marqué, soit un échantillon bien plus gros.
}

\section*{8. Autres applications du TCL}
% ====================================================================

\textbf{Problème 4 --- Pile ou face.}
On lance $n=10\,000$ fois une pièce \textbf{équilibrée} et on note
$S_n$ le nombre de Pile obtenus. On écrit $S_n=X_1+\cdots+X_n$
avec $X_i\sim\mathcal{B}(1/2)$ i.i.d.
\begin{enumerate}
  \item Donner $\E(S_n)$ et $\Var(S_n)$, puis $\sigma(S_n)$.
  \item En déduire, grâce au TCL, l'approximation normale
    de $S_n$. Quelle est la loi de $Z=(S_n-5\,000)/50$\,?
  \item Calculer approximativement $P(S_n\ge 5\,100)$.
    Commenter.
  \item Calculer $P(|S_n-5\,000|\le 100)$.
\end{enumerate}

\atrou[8cm]{%
\begin{enumerate}
  \item $\E(S_n)=n/2=5\,000$ et
    $\Var(S_n)=n\cdot\tfrac{1}{2}\cdot\tfrac{1}{2}=n/4=2\,500$,
    donc $\sigma(S_n)=\sqrt{2500}=50$.
  \item Par le TCL, $S_n\approx\mathcal{N}(5\,000,\,50^2)$, et
    donc $Z=(S_n-5\,000)/50\approx\mathcal{N}(0,1)$.
  \item $P(S_n\ge 5\,100)=P\!\left(Z\ge \tfrac{5100-5000}{50}\right)
    =P(Z\ge 2)=1-\Phi(2)=1-0{,}9772=0{,}0228.$
    Environ $2{,}3\,\%$ de chances d'observer $\ge 5\,100$ Piles sur
    $10\,000$ lancers avec une pièce équilibrée.
  \item $P(|S_n-5\,000|\le 100)=P(|Z|\le 2)=2\Phi(2)-1=0{,}9544.$
    Environ $95\,\%$ des expériences donnent un écart $\le 100$
    (soit $\pm 1\,\%$ autour de la moitié).
\end{enumerate}
}

\medskip

\textbf{Problème 5 --- Un dé suspect.}
On lance un dé à $6$ faces et on compte les $6$ obtenus. À partir
de combien de lancers \emph{sans} aucun $6$ peut-on raisonnablement
soupçonner que le dé est \og truqué \fg\,?
\begin{enumerate}
  \item Sous l'hypothèse \og dé équilibré \fg, donner $\E(S_n)$
    et $\sigma(S_n)$, où $S_n$ est le nombre de $6$ en $n$ lancers.
  \item Combien d'écarts-types sépare \og aucun $6$ \fg{} de la
    moyenne, en fonction de $n$\,?
  \item En déduire l'ordre de grandeur de $n$ à partir duquel
    l'événement \og aucun $6$ \fg{} est à plus de $3\sigma$ de
    la moyenne.
\end{enumerate}

\atrou[6cm]{%
\begin{enumerate}
  \item $S_n\sim$ binomiale $\mathcal{B}(n,1/6)$ :
    $\E(S_n)=n/6$, $\sigma(S_n)=\sqrt{n\cdot\tfrac{1}{6}\cdot\tfrac{5}{6}}
    =\sqrt{5n/36}$.
  \item Écart $\E(S_n)-0=n/6$. Rapporté à $\sigma(S_n)$ :
    \[
      \dfrac{n/6}{\sqrt{5n/36}}=\sqrt{\dfrac{n}{5}}.
    \]
  \item On veut $\sqrt{n/5}\ge 3$, soit $n\ge 45$. Au-delà de
    $n\approx 45$ lancers sans aucun $6$, l'hypothèse \og dé
    équilibré \fg{} devient très peu crédible (moins de
    $0{,}15\,\%$ de chances si elle était vraie).

    \emph{Calcul exact} : $P(\text{aucun }6)=(5/6)^n$ ;
    $(5/6)^{20}\approx 2{,}6\,\%$, $(5/6)^{45}\approx 0{,}02\,\%$.
\end{enumerate}
}

\section*{9. Bilan}
% ====================================================================

\begin{mdframed}[backgroundcolor=ThemeLight, linecolor=Theme,
  linewidth=1.5pt, innertopmargin=10pt, innerbottommargin=10pt]
\textbf{À retenir.}
\begin{enumerate}
  \item Pour $X_1,\dots,X_n$ i.i.d., on note
    $\Xbar = (X_1+\cdots+X_n)/n$.

  \item \textbf{Loi des grands nombres} : $\Xbar\to \E(X_1)$
    quand $n\to +\infty$ (\emph{sans information sur la vitesse}).

  \item \textbf{Loi normale} $\mathcal{N}(\mu,\sigma^2)$ : densité
    en cloche $\tfrac{1}{\sigma\sqrt{2\pi}}e^{-(x-\mu)^2/(2\sigma^2)}$,
    d'espérance $\mu$ et d'écart-type $\sigma$. Sa répartition
    $\Phi$ (pour $\mathcal{N}(0,1)$) est tabulée.

  \item \textbf{Théorème central limite} : pour $n$ grand,
    \[
      \Xbar\approx \E(X_1)+\dfrac{\sigma}{\sqrt{n}}\,Z,
      \quad Z\sim\mathcal{N}(0,1).
    \]
    Vitesse \og en $1/\sqrt{n}$ \fg, loi limite \textbf{universelle}.

  \item \textbf{Standardisation} : $Z=(X-\mu)/\sigma\sim\mathcal{N}(0,1)$
    ramène tout calcul à la table de $\Phi$.

  \item \textbf{Réflexes de table} : $\Phi(-a)=1-\Phi(a)$,
    $P(a\le Z\le b)=\Phi(b)-\Phi(a)$, $P(|Z|\le a)=2\Phi(a)-1$.

  \item \textbf{Règle 68 / 95 / 99} :
    $P(|Z|\le 1)\approx 0{,}68$, $P(|Z|\le 1{,}96)=0{,}95$,
    $P(|Z|\le 2{,}58)=0{,}99$.

  \item \textbf{Sondage binaire de taille $n$} : marge d'erreur
    $\pm 1/\sqrt{n}$ à $95\,\%$ de confiance.
\end{enumerate}
\end{mdframed}
