\section*{1 -- Une densité à apprivoiser}
% ====================================================================

Sur un ouvrage de génie civil, on mesure la longueur (en mm) d'une
micro-fissure apparue au bout de quelques mois.
On modélise cette longueur par une variable continue $X$ de densité
\[
  f(x)=
  \begin{cases}
    c\,x(3-x) & \text{si } x\in[0,3],\\
    0 & \text{sinon.}
  \end{cases}
\]

\begin{enumerate}
  \item Tracer rapidement l'allure de $f$ sur $[0,3]$.
  Où semble concentrée la masse de probabilité ?
  \begin{solution}
    \textit{Piste.} Étudier brièvement la fonction $x\mapsto x(3-x)$ :
    quelles sont ses racines ? Pourquoi la courbe est-elle symétrique,
    et autour de quelle valeur ? Où se situe son maximum ?
    Reliez cette allure à la notion de \og zone de concentration \fg{}
    de la probabilité.
  \end{solution}

  \item Déterminer la constante $c$ pour que $f$ soit une densité.
  \begin{solution}
    \textit{Méthode.} Deux conditions caractérisent une densité :
    $f\ge 0$ (que dire du signe de $c$ ?) et
    $\int_{\mathbb R}f=1$. Ici, l'intégrale se réduit à
    $\int_0^3 c\,x(3-x)\,dx$ : développez $x(3-x)=3x-x^2$ avant
    d'intégrer, puis isolez $c$.
  \end{solution}

  \item Rappeler ce que représente l'aire sous
  la courbe de $f$ entre $a$ et $b$.
  \begin{solution}
    \textit{Indice.} Revoyez la définition d'une densité : l'aire
    sous la courbe correspond à une probabilité associée à un
    événement de la forme $\{a\le X\le b\}$. Écrivez clairement
    cet événement.
  \end{solution}

  \item Quelle est la probabilité que la fissure mesure moins
  de $1$~mm ? Plus de $2$~mm ?
  \begin{solution}
    \textit{Piste.} Traduisez chaque probabilité par une intégrale
    $\int_a^b f(t)\,dt$ bien choisie. Pour aller vite, commencez
    par calculer la fonction de répartition
    $F(x)=\int_0^x f(t)\,dt$ sur $[0,3]$ (une primitive de
    $t(3-t)$ suffit) ; exprimez ensuite $P(X\le 1)$ et $P(X>2)$
    à l'aide de $F$. Une fois vos résultats obtenus, comparez-les :
    la symétrie de $f$ autour de $3/2$ vous donne-t-elle un résultat
    cohérent ?
  \end{solution}

  \item Comparer, sans calcul, les deux probabilités
  $P(X\in[1{,}4\,;\,1{,}6])$ et $P(X\in[0{,}1\,;\,0{,}3])$.
  Que dit ce résultat sur l'information qu'apporte la densité ?
  \begin{solution}
    \textit{Indice.} Comparez la \emph{longueur} des deux intervalles,
    puis la \emph{valeur} de $f$ sur chacun (grâce à l'allure
    tracée en Q1). Rappelez-vous qu'intuitivement
    $P(X\in[x_0,x_0+\varepsilon])\approx f(x_0)\,\varepsilon$ pour
    un petit intervalle : que dit alors la comparaison ?
    Quelle analogie physique (masse, densité) peut-on faire ?
  \end{solution}

  \item Calculer $E[X]$ et interpréter.
  \begin{solution}
    \textit{Méthode.} Appliquez la définition
    $E[X]=\int_{\mathbb R} x\,f(x)\,dx$ (ici l'intégrale se limite à
    $[0,3]$). Développez $x\cdot x(3-x)$ avant d'intégrer.
    \textit{Contrôle.} Pouvez-vous prévoir la valeur
    sans calcul grâce à la symétrie de $f$ vue en Q1 ?
  \end{solution}
\end{enumerate}

\section*{2 -- Fonction de répartition : aller-retour avec la densité}
% ====================================================================

On rappelle que
\[
  F_X(x)=P(X\le x)=\int_{-\infty}^x f_X(t)\,dt
  \qquad\text{et}\qquad
  f_X=F_X'\text{ là où }F_X\text{ est dérivable.}
\]

\begin{enumerate}
  \item Exprimer $P(X>x)$ puis $P(a\le X\le b)$ en fonction de $F_X$.
  Pourquoi la continuité de $X$ permet-elle d'ignorer la différence
  entre inégalités strictes et larges ?
  \begin{solution}
    \textit{Piste.} Les événements $\{X>x\}$ et $\{X\le x\}$ sont
    complémentaires : que vaut la somme de leurs probabilités ?
    Pour $\{a\le X\le b\}$, écrivez-le comme différence de deux
    événements du type $\{X\le\cdot\}$.

    Pour la dernière question, que vaut $P(X=a)=\int_a^a f$ ?
    Conclusion sur $P(X\le a)$ et $P(X<a)$.
  \end{solution}

  \item \textbf{De $f$ vers $F$.}
  On considère la densité
  $f(x)=\tfrac{3}{8}x^2$ pour $x\in[0,2]$, et $0$ sinon
  (par exemple la loi d'une variable qui se concentre vers les
  grandes valeurs).
  \begin{enumerate}
    \item Vérifier que $f$ est une densité.
    \item Déterminer $F$ sur $\mathbb R$.
    \item Calculer $P(X\le 1)$ et $P(X>1{,}5)$.
  \end{enumerate}
  \begin{solution}
    \textit{Pistes.}
    \begin{enumerate}
    \item Vérifiez les deux conditions habituelles : signe de $f$
    et valeur de $\int_0^2 f$.
    \item Discutez selon la position de $x$ : en dehors de $[0,2]$,
    $f$ est nulle (que vaut alors $F$ ?). Sur $[0,2]$,
    primitivez $\tfrac{3}{8}t^2$. N'oubliez pas d'écrire $F$ sur les
    trois zones ($x<0$, $x\in[0,2]$, $x>2$).
    \item Exprimez chaque probabilité à l'aide de $F$ (cf. Q1).
    \end{enumerate}
  \end{solution}
  \newpage

  \item \textbf{De $F$ vers $f$.}
  On donne maintenant la fonction de répartition
  du délai $D$ (en jours) d'une livraison~:
  \[
    F_D(t)=
    \begin{cases}
      0 & \text{si } t<0,\\
      1-\bigl(1-t/5\bigr)^3 & \text{si } t\in[0,5],\\
      1 & \text{si } t>5.
    \end{cases}
  \]
  \begin{enumerate}
    \item Vérifier que $F_D$ a bien les propriétés attendues
    d'une fonction de répartition (positivité, croissance, limites).
    \item En déduire la densité $f_D$.
    \item Calculer $P(D>2)$, $P(1\le D\le 3)$,
    puis la \og médiane \fg, c'est-à-dire l'instant $t_{1/2}$ pour lequel
    $P(D\le t_{1/2})=1/2$.
  \end{enumerate}
  \begin{solution}
    \textit{Pistes.}
    \begin{enumerate}
    \item Listez ce qu'on attend d'une fonction de répartition :
    continuité, monotonie, limites en $\pm\infty$. Vérifiez-les
    une à une (la monotonie se lit sur le signe de $F_D'$).
    \item Rappel : là où $F$ est dérivable, $f=F'$. Dérivez
    $(1-t/5)^3$ avec la règle de la chaîne.
    \item Exprimez chaque probabilité à l'aide de $F_D$.
    Pour la médiane, posez l'équation $F_D(t_{1/2})=1/2$ et
    isolez $t_{1/2}$ (attention à la racine cubique).

    \textit{Question de sens.} La densité $f_D$ est-elle
    croissante ou décroissante ? Cela explique-t-il que la médiane
    soit petite, alors que $D$ peut atteindre $5$ ?
    \end{enumerate}
  \end{solution}

  \item 
  Peut-on avoir une fonction de répartition $F$ telle que $F(3)=0{,}8$
  et $F(2)=0{,}9$ ?
  \begin{solution}
    \textit{Indice.} Quelle propriété fondamentale une fonction de
    répartition vérifie-t-elle toujours (pensez au signe de $f$ et
    à $F(x)=\int_{-\infty}^x f$) ? Confrontez cette propriété aux
    valeurs proposées.
  \end{solution}
\end{enumerate}

\section*{3 -- Espérance, variance, propriétés}
% ====================================================================

On rappelle :
\[
  E[X]=\int_{-\infty}^{+\infty}\!x\,f_X(x)\,dx,\qquad
  E[g(X)]=\int_{-\infty}^{+\infty}\!g(x)\,f_X(x)\,dx,
\]
\[
  \text{Var}(X)=E\bigl[(X-E[X])^2\bigr]=E[X^2]-E[X]^2.
\]

\begin{enumerate}
  \item \textbf{Une densité constante.}
  Soit $X$ dont la densité est constante
  sur $[a,b]$, et $0$ sinon (on \og tire au hasard uniformément \fg{}
  une valeur dans $[a,b]$).
  \begin{enumerate}
    \item Explicitez la fonction de densité de $X$. L'ecrire avec une fonction indicatrice. 
    \item Calculer $E[X]$.
    \item Calculer $E[X^2]$, puis $\text{Var}(X)$.
  \end{enumerate}
  \begin{solution}
    \textit{Pistes.}
    \begin{enumerate}
    \item La densité est une constante $k$ sur $[a,b]$ :
    déterminez $k$ grâce à la condition $\int f=1$. La notation
    indicatrice $\mathbf{1}_{[a,b]}(x)$ vaut $1$ sur l'intervalle
    et $0$ ailleurs.
    \item Appliquez la définition de $E[X]$ et sortez la constante
    de l'intégrale. Anticipez le résultat par symétrie.
    \item Même principe pour $E[X^2]$. Utilisez la formule de
    König : $\text{Var}(X)=E[X^2]-E[X]^2$, puis simplifiez
    (factorisation de $b^3-a^3$ et réduction au même dénominateur).
    \end{enumerate}
  \end{solution}

  \item Reprenons la densité parabolique $f(x)=\tfrac{2}{9}x(3-x)$
  sur $[0,3]$ de l'exercice~1. On a déjà montré que $E[X]=3/2$.
  Calculer $\text{Var}(X)$ et l'écart-type $\sigma(X)$.
  Comparer avec la variance d'une densité constante sur $[0,3]$ ;
  commenter.
  \begin{solution}
    \textit{Méthode.} Calculez d'abord $E[X^2]$ en appliquant la
    définition : $\int_0^3 x^2\,f(x)\,dx$, après avoir développé
    $x^2\cdot x(3-x)$. Utilisez ensuite $\text{Var}(X)=E[X^2]-E[X]^2$
    (la valeur $E[X]=3/2$ est déjà connue). L'écart-type s'en
    déduit par $\sigma(X)=\sqrt{\text{Var}(X)}$.

    \textit{Comparaison.} Calculez la variance de la loi constante
    sur $[0,3]$ à l'aide de la formule de la question précédente,
    puis comparez les deux : laquelle des deux densités vous paraît
    la plus \og étalée \fg{} (revoir l'allure de $f$) ? Le résultat
    est-il cohérent ?
  \end{solution}

  \item \textbf{Théorème de transfert.}
  Reprenons la même densité $f(x)=\tfrac{2}{9}x(3-x)$ sur $[0,3]$.
  On définit $Y=X(3-X)$ (par exemple, une énergie proportionnelle
  à $X(3-X)$).
  Calculer $E[Y]$ \emph{sans} déterminer la loi de~$Y$.
  \begin{solution}
    \textit{Piste.} Le théorème de transfert dit que
    $E[g(X)]=\int g(x)\,f_X(x)\,dx$, sans passer par la loi de $Y$.
    Ici $g(x)=x(3-x)$, donc on intègre $g(x)\cdot f(x)$ :
    vous obtenez $\tfrac{2}{9}\int_0^3 (x(3-x))^2\,dx$.
    Développez $(x(3-x))^2=9x^2-6x^3+x^4$ avant d'intégrer.

    \textit{À méditer.} Essayez d'imaginer comment vous feriez
    pour déterminer d'abord la loi de $Y$, puis $E[Y]$ par
    sa définition : vous verrez que c'est bien plus long.
  \end{solution}

  \item \textbf{Propriété sur les transformations affines.}
  On suppose que $E[X]=\mu$ et $\text{Var}(X)=\sigma^2$,
  et on pose $Y=aX+b$ avec $a\in\mathbb R$, $b\in\mathbb R$.
  Exprimer $E[Y]$, $\text{Var}(Y)$ et $\sigma(Y)$.
  Que se passe-t-il si $a<0$ ?
  \begin{solution}
    \textit{Méthode.} Utilisez la linéarité de l'espérance pour
    $E[Y]$. Pour la variance, partez de la définition
    $\text{Var}(Y)=E[(Y-E[Y])^2]$ et factorisez $(aX+b-(a\mu+b))^2$
    par $a^2$. Attention au passage à l'écart-type :
    $\sqrt{a^2}=|a|$ (et non $a$).

    \textit{Sens physique.} Essayez avec $X$ en mètres et $Y=100X$
    en cm : les deux quantités (moyenne et écart-type) doivent-elles
    être multipliées par le même facteur ? Pourquoi ?
  \end{solution}
\end{enumerate}

\section*{4 -- Densité d'une fonction de variable aléatoire}
% ====================================================================

Dans cet exercice, on appliquera les
méthodes du CM2 pour obtenir la densité
de $Y=g(X)$ 

\subsection*{4.a -- Transformation affine : $Y=aX+b$}

Soit $X$ une variable continue de densité $f_X$ et de fonction
de répartition $F_X$. On pose $Y=aX+b$ avec $a>0$.

\begin{enumerate}
  \item Exprimer $F_Y(y)$ en fonction de $F_X$.
  \begin{solution}
    \textit{Piste.} Partez de la définition
    $F_Y(y)=P(Y\le y)=P(aX+b\le y)$ et isolez $X$ dans l'inégalité.
    Comme $a>0$, le sens de l'inégalité est conservé. Reconnaissez
    alors une probabilité du type $P(X\le \cdot)$ que l'on réécrit
    avec $F_X$.
  \end{solution}

  \item En déduire la densité $f_Y$.
  \begin{solution}
    \textit{Méthode.} Dérivez l'expression de $F_Y$ trouvée
    à la question précédente, en appliquant la règle de dérivation
    des fonctions composées (attention au facteur qui vient
    de la dérivée de $\tfrac{y-b}{a}$).

    \textit{À chercher ensuite.} Que deviendrait la formule
    si $a<0$ ? (Indice : il faut alors inverser le sens de
    l'inégalité et un $|a|$ apparaît naturellement.)
  \end{solution}

  \item \textbf{Application numérique.}
  La longueur d'une pièce (en mm) est modélisée par $X$ de densité
  $f_X(x)=\tfrac{3}{8}x^2$ sur $[0,2]$. On choisit de l'exprimer en cm :
  $Y=X/10$. Déterminer la densité de $Y$.
  \begin{solution}
    \textit{Piste.} Identifiez d'abord les valeurs de $a$ et $b$
    correspondant à $Y=X/10$. Appliquez ensuite la formule de la
    question précédente, sans oublier de déterminer le \emph{nouvel
    intervalle} sur lequel $Y$ prend ses valeurs (si $X\in[0,2]$,
    que vaut $Y$ ?).

    \textit{Contrôle.} Vérifiez que $\int f_Y=1$ sur ce nouvel
    intervalle.
  \end{solution}
\end{enumerate}

\subsection*{4.b -- Passage au carré : $Y=X^2$}

Soit $X$ de densité
\[
  f_X(x)=x\,e^{-x^2/2}\quad\text{pour }x\ge 0,\qquad
  f_X(x)=0\quad\text{sinon.}
\]

\begin{enumerate}
  \item Vérifier que $f_X$ est bien une densité.
  \begin{solution}
    \textit{Indice.} Positivité : immédiate. Pour l'intégrale,
    reconnaissez une forme du type $u'\,e^u$ avec $u(x)=-x^2/2$
    (ou posez le changement de variable $t=x^2/2$). Une primitive
    apparaît naturellement.
  \end{solution}

  \item On pose $Y=X^2$. Quel intervalle de valeurs prend $Y$ ?
  Pour $y\ge 0$, écrire $\{Y\le y\}$ comme un événement portant
  sur $X$.
  \begin{solution}
    \textit{Piste.} Déterminez d'abord le support de $Y$ :
    sachant que $X\ge 0$, où se situe $X^2$ ? Pour l'événement
    $\{Y\le y\}$, traduisez l'inégalité $X^2\le y$ en pensant
    à la fonction racine carrée (attention : cela marche
    uniquement parce que $X\ge 0$).
  \end{solution}

  \item En déduire $F_Y(y)$ sous forme d'intégrale.
  \begin{solution}
    \textit{Indice.} D'après la question précédente,
    $F_Y(y)=P(Y\le y)=P(X\le\sqrt{y})$ pour $y\ge 0$. Écrivez cette
    probabilité comme une intégrale de $f_X$ entre deux bornes
    bien choisies. Que vaut $F_Y$ pour $y<0$ (support de $Y$) ?
  \end{solution}

  \item Effectuer le \textbf{changement de variable} $u=x^2$
  dans cette intégrale (de sorte que $du=2x\,dx$) et en déduire
  l'expression de $F_Y$, puis de $f_Y$.
  \begin{solution}
    \textit{Méthode.}
    \begin{itemize}
    \item Posez $u=x^2$. Que deviennent les bornes lorsque $x$ varie
    de $0$ à $\sqrt{y}$ ? Comment $x\,dx$ se réécrit-il en fonction
    de $du$ ?
    \item Après substitution, l'intégrale devient une intégrale
    classique de $e^{-u/2}$ que vous savez primitiver.
    \item Pour $f_Y$, rappelez-vous que $f=F'$ là où $F$ est dérivable,
    et n'oubliez pas de préciser $f_Y$ en-dehors de $[0,+\infty[$.
    \end{itemize}
    \textit{Vérification.} La densité obtenue doit être positive
    et d'intégrale $1$.
  \end{solution}

  \item Calculer $E[Y]$ de deux manières différentes~:
  \begin{enumerate}
    \item directement avec la densité $f_Y$ trouvée ci-dessus ;
    \item avec le théorème de transfert appliqué à $X$,
  sous la forme $E[Y]=E[X^2]$.
  \end{enumerate}
  \begin{solution}
    \textit{Pistes.}
    \begin{enumerate}
    \item Avec $f_Y$ : il s'agit de calculer
    $\int_0^{+\infty}y\,f_Y(y)\,dy$. Une \emph{intégration par
    parties} avec $u=y$, $v'=\tfrac{1}{2}e^{-y/2}$ ramène à une
    intégrale standard. (Pensez à utiliser
    $\int_0^{+\infty}e^{-y/2}\,dy=2$.)
    \item Par transfert : calculez $E[X^2]=\int_0^{+\infty}x^2\,f_X(x)\,dx
    =\int_0^{+\infty}x^3\,e^{-x^2/2}\,dx$. Le changement de variable
    $u=x^2/2$ transforme cette intégrale en
    $\int_0^{+\infty}2u\,e^{-u}\,du$ (une intégrale classique liée
    à $\Gamma(2)$).
    \end{enumerate}
    Les deux méthodes doivent bien sûr donner la même valeur.
  \end{solution}
\end{enumerate}

\subsection*{4.c -- Transformation logarithmique : $Y=-\ln X$}

Soit $X$ de densité constante $f_X(x)=1$ sur $[0,1]$, et $0$ sinon.
On pose $Y=-\ln X$.

\begin{enumerate}
  \item Justifier que $Y$ prend ses valeurs dans $[0,+\infty[$.
  \begin{solution}
    \textit{Piste.} Rappel : sur quel intervalle $\ln x$ est-il
    négatif, nul, positif ? $X$ prend ses valeurs dans $]0,1]$ :
    quels sont alors les signes et les limites de $-\ln X$ ?
  \end{solution}

  \item Pour $y\ge 0$, écrire $\{Y\le y\}$ comme un événement portant
  sur $X$, puis en déduire $F_Y(y)$.
  \begin{solution}
    \textit{Méthode.} Partez de $\{-\ln X\le y\}$ et manipulez
    l'inégalité : multipliez par $-1$ (changement de sens) puis
    appliquez l'exponentielle (croissante). Vous obtenez un
    événement de la forme $\{X\ge c\}$ avec $c$ dépendant de $y$.

    Comme $f_X=1$ sur $[0,1]$, la probabilité $P(X\ge c)$ se
    calcule simplement comme une aire rectangulaire, d'où
    l'expression de $F_Y(y)$ pour $y\ge 0$.
  \end{solution}

  \item En déduire la densité de $Y$.
  \begin{solution}
    \textit{Indice.} $f_Y=F_Y'$ là où $F_Y$ est dérivable.
    N'oubliez pas de préciser $f_Y$ en-dehors du support de $Y$.

    \textit{Pour aller plus loin.} La loi obtenue correspond à celle
    des \og temps d'attente sans mémoire \fg{} : c'est exactement
    comme cela que la plupart des langages de programmation
    simulent une telle loi à partir d'un tirage uniforme sur $[0,1]$.
  \end{solution}

  \item Calculer $E[Y]$ par deux méthodes~:
  \begin{enumerate}
    \item via la densité $f_Y$ ;
    \item via le théorème de transfert
    $E[Y]=E[-\ln X]=\int_0^1(-\ln x)\,dx$.
  \end{enumerate}
  \begin{solution}
    \textit{Pistes.}
    \begin{enumerate}
    \item Via $f_Y$ : calculez $\int_0^{+\infty}y\,e^{-y}\,dy$
    par \emph{intégration par parties} avec $u=y$, $v'=e^{-y}$.
    Attention au terme tout intégré en $+\infty$ (qui tend vers $0$).
    \item Par transfert : calculez $\int_0^1(-\ln x)\,dx$,
    également par IPP (avec $u=\ln x$). Vous aurez besoin de la
    limite classique $\lim_{x\to 0^+}x\ln x=0$.
    \end{enumerate}
    Les deux calculs doivent converger vers la même valeur : c'est
    un bon moyen de vérifier qu'aucune erreur ne s'est glissée !
  \end{solution}
\end{enumerate}

\section*{5 -- Application}
Dans une station-service, la demande hebdomadaire en essence, en milliers de litres, est une variable aléatoire $X$ de densité $f(x)=c(1-x)^4 \mathbf{1}_{[0,1]}$.
\begin{enumerate}
  \item Déterminer $c$.

  \item La station est réapprovisionnée chaque lundi à 20 h . Quelle doit être la capacité du réservoir d'essence pour que la probabilité d'épuiser ce réservoir soit inférieure à $10^{-5}$ ?

\end{enumerate}

\section*{6 -- Manipulation de densités}
Soit $f$ la fonction de $\mathbb{R}$ dans $\mathbb{R}$ définie par

$$
f(x)=\frac{e^{-x}}{\left(1+e^{-x}\right)^2}
