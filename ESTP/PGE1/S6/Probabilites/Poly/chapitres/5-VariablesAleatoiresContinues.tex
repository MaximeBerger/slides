% Chapitre 4 : Variables aléatoires continues

\section{Quand utiliser une variable continue ?}

Une variable est \textbf{continue} lorsqu'elle représente une grandeur mesurée :
\begin{itemize}
    \item résistance d'une éprouvette ;
    \item épaisseur d'une couche d'enrobé ;
    \item temps d'attente d'une livraison ;
    \item hauteur de pluie ;
    \item longueur d'une fissure.
\end{itemize}

\begin{Rmq}
Pour une variable continue, la probabilité d'obtenir exactement une valeur donnée est nulle. On calcule des probabilités sur des \textbf{intervalles}.
\end{Rmq}

\section{Densité de probabilité}

\begin{Def}
Une variable continue $X$ admet une \textbf{densité} $f$ si, pour tout intervalle $[a,b]$,
$$
\mathbb{P}(a\leq X\leq b)=\int_a^b f(x)\,dx.
$$
\end{Def}

\begin{Prop}
Une densité vérifie :
\begin{itemize}
    \item $f(x)\geq 0$ pour tout $x$ ;
    \item $\int_{-\infty}^{+\infty} f(x)\,dx=1$.
\end{itemize}
\end{Prop}

\begin{Meth}
Pour vérifier qu'une fonction est une densité, on vérifie toujours :
\begin{enumerate}
    \item la positivité ;
    \item l'intégrale totale égale à $1$.
\end{enumerate}
\end{Meth}

\section{Fonction de répartition}

\begin{Def}
La \textbf{fonction de répartition} de $X$ est
$$
F_X(x)=\mathbb{P}(X\leq x).
$$
\end{Def}

\begin{Prop}
Si $X$ admet une densité $f$, alors
$$
F_X(x)=\int_{-\infty}^x f(t)\,dt.
$$
Et pour $a<b$ :
$$
\mathbb{P}(a<X\leq b)=F_X(b)-F_X(a).
$$
\end{Prop}

\begin{Rmq}
Selon l'énoncé, on peut travailler soit avec la densité, soit avec la fonction de répartition. Il faut choisir l'écriture la plus simple pour le calcul demandé.
\end{Rmq}

\section{Espérance et variance}

\begin{Def}
L'espérance de $X$ est
$$
\mathbb{E}[X]=\int_{-\infty}^{+\infty} x f(x)\,dx.
$$
\end{Def}

\begin{Def}
La variance de $X$ est
$$
\Var(X)=\mathbb{E}[X^2]-\mathbb{E}[X]^2.
$$
\end{Def}

\begin{Def}
L'écart-type de $X$ est
$$
\sigma(X)=\sqrt{\Var(X)}.
$$
\end{Def}

\begin{Rmq}
L'espérance donne une valeur moyenne théorique ; l'écart-type mesure la dispersion autour de cette valeur.
\end{Rmq}

\section{Lois continues à connaître}

\subsection{Loi uniforme}

\begin{Def}
On dit que $X$ suit une \textbf{loi uniforme} sur $[a,b]$ si sa densité est
$$
f(x)=
\begin{cases}
\frac{1}{b-a} & \text{si } x\in[a,b],\\
0 & \text{sinon}.
\end{cases}
$$
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{U}([a,b])$, alors
$$
\mathbb{E}[X]=\frac{a+b}{2},
\qquad
\Var(X)=\frac{(b-a)^2}{12}.
$$
\end{Prop}

\begin{Ex}
Modèle simplifié pour une mesure répartie uniformément dans une tolérance.
\end{Ex}

\subsection{Loi exponentielle}

\begin{Def}
On dit que $X$ suit une \textbf{loi exponentielle} de paramètre $\lambda>0$ si
$$
f(x)=
\begin{cases}
\lambda e^{-\lambda x} & \text{si } x\geq 0,\\
0 & \text{sinon}.
\end{cases}
$$
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{E}(\lambda)$, alors
$$
\mathbb{E}[X]=\frac{1}{\lambda},
\qquad
\Var(X)=\frac{1}{\lambda^2},
\qquad
F_X(x)=1-e^{-\lambda x} \ \text{pour } x\geq 0.
$$
\end{Prop}

\begin{Ex}
Loi adaptée à un temps d'attente ou à une durée de vie simplifiée lorsque l'on souhaite un modèle simple sans mémoire.
\end{Ex}

\subsection{Loi normale}

\begin{Def}
Une variable $X$ suit une \textbf{loi normale} de moyenne $\mu$ et de variance $\sigma^2$, notée $\mathcal{N}(\mu,\sigma^2)$, si sa densité est :
$$
f(x)=\frac{1}{\sigma\sqrt{2\pi}}\exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right).
$$
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{N}(\mu,\sigma^2)$, alors
$$
\mathbb{E}[X]=\mu,
\qquad
\Var(X)=\sigma^2.
$$
\end{Prop}

\begin{Rmq}
La loi normale est très utilisée en ingénierie pour modéliser des mesures influencées par de nombreux petits effets : résistance, dimensions, erreurs de mesure, délais.
\end{Rmq}

\subsection{Centrage-réduction}

\begin{Prop}
Si $X\sim \mathcal{N}(\mu,\sigma^2)$, alors
$$
Z=\frac{X-\mu}{\sigma}
$$
suit la loi normale centrée réduite $\mathcal{N}(0,1)$.
\end{Prop}

\begin{Meth}
Pour calculer une probabilité avec une loi normale :
\begin{enumerate}
    \item on transforme $X$ en $Z=\dfrac{X-\mu}{\sigma}$ ;
    \item on utilise la table de la loi normale centrée réduite ;
    \item on revient à la question de départ.
\end{enumerate}
\end{Meth}

\section{Interprétation des paramètres}

\begin{itemize}
    \item $\mu$ est le niveau moyen attendu.
    \item $\sigma$ mesure la dispersion.
    \item Une grande valeur de $\sigma$ signifie un procédé peu stable.
\end{itemize}

\begin{Ex}
\textbf{Contrôle de résistance}

Si la résistance suit une loi normale de moyenne $33$ MPa et d'écart-type $3$ MPa, alors la moyenne est satisfaisante, mais la dispersion doit encore être comparée aux exigences de conformité.
\end{Ex}

\section{Erreurs fréquentes}

\begin{itemize}
    \item Oublier de vérifier qu'une densité est positive et d'intégrale $1$.
    \item Calculer $\mathbb{P}(X=x)$ pour une variable continue.
    \item Confondre densité et probabilité.
    \item Utiliser une loi normale sans avoir interprété ses paramètres.
\end{itemize}

\section{Ce qu'il faut savoir faire}

\begin{itemize}
    \item vérifier qu'une fonction est une densité ;
    \item calculer une probabilité sur un intervalle ;
    \item passer de la densité à la fonction de répartition ;
    \item reconnaître les lois uniforme, exponentielle et normale ;
    \item utiliser le centrage-réduction pour une loi normale.
\end{itemize}
