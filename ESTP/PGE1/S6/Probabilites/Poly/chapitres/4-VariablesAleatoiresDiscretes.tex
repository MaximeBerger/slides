% Chapitre 3 : Variables aléatoires discrètes

\section{Quand utiliser une variable discrète ?}

Une variable aléatoire est \textbf{discrète} lorsqu'elle compte un nombre de réalisations :
\begin{itemize}
    \item nombre de pièces défectueuses dans un lot ;
    \item nombre de jours de pluie dans une semaine ;
    \item nombre d'incidents sur une période ;
    \item nombre d'éprouvettes non conformes parmi $n$ essais.
\end{itemize}

\section{Loi d'une variable aléatoire discrète}

\begin{Def}
Si $X$ prend les valeurs $x_1,\dots,x_n$ ou une suite dénombrable de valeurs, sa \textbf{loi} est la liste des probabilités
$$
\mathbb{P}(X=x_i).
$$
\end{Def}

\begin{Prop}
Les probabilités d'une loi discrète vérifient :
\begin{itemize}
    \item $\mathbb{P}(X=x_i)\geq 0$ ;
    \item $\sum_i \mathbb{P}(X=x_i)=1$.
\end{itemize}
\end{Prop}

\begin{Ex}
\textbf{Nombre d'éprouvettes non conformes}

On prélève $6$ éprouvettes sur un lot. La variable
$$
X = \text{nombre d'éprouvettes non conformes}
$$
prend les valeurs $0,1,2,3,4,5,6$.
\end{Ex}

\section{Fonction de répartition}

\begin{Def}
La \textbf{fonction de répartition} de $X$ est
$$
F_X(x)=\mathbb{P}(X\leq x).
$$
\end{Def}

\begin{Rmq}
La fonction de répartition permet de calculer rapidement des probabilités du type :
\begin{itemize}
    \item $\mathbb{P}(X\leq a)$ ;
    \item $\mathbb{P}(a<X\leq b)=F_X(b)-F_X(a)$.
\end{itemize}
\end{Rmq}

\section{Espérance, variance et écart-type}

\begin{Def}
L'\textbf{espérance} de $X$ est :
$$
\mathbb{E}[X]=\sum_i x_i \mathbb{P}(X=x_i).
$$
\end{Def}

\begin{Rmq}
L'espérance représente une \textbf{valeur moyenne théorique}. Elle n'est pas forcément une valeur possible de la variable.
\end{Rmq}

\begin{Def}
La \textbf{variance} de $X$ est :
$$
\Var(X)=\mathbb{E}\left[(X-\mathbb{E}[X])^2\right]
=\mathbb{E}[X^2]-\mathbb{E}[X]^2.
$$
\end{Def}

\begin{Def}
L'\textbf{écart-type} de $X$ est :
$$
\sigma(X)=\sqrt{\Var(X)}.
$$
\end{Def}

\begin{Meth}
Dans un problème appliqué :
\begin{itemize}
    \item l'espérance sert souvent à évaluer un \textbf{coût moyen}, un \textbf{nombre moyen} ou une \textbf{production moyenne} ;
    \item la variance et l'écart-type mesurent la \textbf{dispersion} autour de cette moyenne.
\end{itemize}
\end{Meth}

\begin{Prop}[Linéarité de l'espérance]
Pour $a,b\in \mathbb{R}$ :
$$
\mathbb{E}[aX+b]=a\mathbb{E}[X]+b.
$$
\end{Prop}

\begin{Prop}
Si $X$ et $Y$ sont indépendantes :
$$
\mathbb{E}[X+Y]=\mathbb{E}[X]+\mathbb{E}[Y]
\qquad \text{et} \qquad
\Var(X+Y)=\Var(X)+\Var(Y).
$$
\end{Prop}

\section{Fonction indicatrice}

\begin{Def}
Si $A$ est un événement, sa \textbf{fonction indicatrice} est
$$
\mathbf{1}_A=
\begin{cases}
1 & \text{si } A \text{ est réalisé},\\
0 & \text{sinon}.
\end{cases}
$$
\end{Def}

\begin{Prop}
Pour une indicatrice :
$$
\mathbb{E}[\mathbf{1}_A]=\mathbb{P}(A), \qquad
\Var(\mathbf{1}_A)=\mathbb{P}(A)\bigl(1-\mathbb{P}(A)\bigr).
$$
\end{Prop}

\begin{Rmq}
Compter un nombre d'événements revient souvent à sommer des indicatrices. C'est l'idée de base derrière la loi binomiale.
\end{Rmq}

\section{Lois discrètes à connaître}

\subsection{Loi de Bernoulli}

\begin{Def}
Une variable $X$ suit une \textbf{loi de Bernoulli} de paramètre $p$ si :
$$
\mathbb{P}(X=1)=p, \qquad \mathbb{P}(X=0)=1-p.
$$
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{B}(p)$, alors :
$$
\mathbb{E}[X]=p, \qquad \Var(X)=p(1-p).
$$
\end{Prop}

\begin{Ex}
Sur une pièce contrôlée, on note $X=1$ si la pièce est conforme, $X=0$ sinon.
\end{Ex}

\subsection{Loi binomiale}

\begin{Def}
Une variable $X$ suit une \textbf{loi binomiale} de paramètres $n$ et $p$, notée $\mathcal{B}(n,p)$, si elle compte le nombre de succès dans $n$ essais indépendants de même probabilité de succès $p$.
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{B}(n,p)$, alors pour $k\in\{0,\dots,n\}$ :
$$
\mathbb{P}(X=k)=\binom{n}{k}p^k(1-p)^{n-k}.
$$
De plus :
$$
\mathbb{E}[X]=np, \qquad \Var(X)=np(1-p).
$$
\end{Prop}

\begin{Ex}
On contrôle $20$ éprouvettes, chacune ayant une probabilité $p=0.03$ d'être non conforme, de façon indépendante. Le nombre $X$ d'éprouvettes non conformes suit une loi binomiale $\mathcal{B}(20,0.03)$.
\end{Ex}

\subsection{Loi géométrique}

\begin{Def}
Une variable $T$ suit une \textbf{loi géométrique} de paramètre $p \in ]0,1]$ si elle modélise le rang du premier succès dans une suite d'essais de Bernoulli indépendants de probabilité de succès $p$.
\end{Def}

\begin{Prop}
Si $T$ suit une loi géométrique de paramètre $p$, alors pour tout $k \geq 1$ :
$$
\mathbb{P}(T=k)=(1-p)^{k-1}p.
$$
De plus :
$$
\mathbb{E}[T]=\frac{1}{p}.
$$
\end{Prop}

\begin{Ex}
Si chaque pièce contrôlée a une probabilité $p=0.04$ d'être défectueuse, le rang du premier défaut suit une loi géométrique de paramètre $0.04$.
\end{Ex}

\begin{Rmq}
Cette loi est utile lorsqu'on ne compte pas le \textbf{nombre total} de défauts, mais le \textbf{temps d'attente discret} avant le premier défaut, le premier incident ou le premier succès.
\end{Rmq}

\subsection{Loi de Poisson}

\begin{Def}
Une variable $X$ suit une \textbf{loi de Poisson} de paramètre $\lambda>0$ si :
$$
\mathbb{P}(X=k)=e^{-\lambda}\frac{\lambda^k}{k!}, \qquad k\in \mathbb{N}.
$$
\end{Def}

\begin{Prop}
Si $X\sim \mathcal{P}(\lambda)$, alors :
$$
\mathbb{E}[X]=\lambda, \qquad \Var(X)=\lambda.
$$
\end{Prop}

\begin{Ex}
La loi de Poisson est adaptée pour modéliser un \textbf{nombre d'événements rares} :
\begin{itemize}
    \item nombre de pannes sur une semaine ;
    \item nombre de defauts sur une grande surface ;
    \item nombre d'appels d'alerte par heure.
\end{itemize}
\end{Ex}

\section{Choisir la bonne loi}

\begin{Meth}
\begin{itemize}
    \item \textbf{Bernoulli} : un seul essai, succès ou échec.
    \item \textbf{Binomiale} : répétition de $n$ essais indépendants identiques.
    \item \textbf{Géométrique} : rang du premier succès dans des essais indépendants identiques.
    \item \textbf{Poisson} : comptage d'événements rares sur une durée, une longueur, une surface ou un volume.
\end{itemize}
\end{Meth}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|m{4cm}|m{4.3cm}|m{5cm}|}
\hline
\textbf{Situation typique} & \textbf{Loi conseillée} & \textbf{Question à se poser} \\
\hline
Un seul contrôle, succès ou échec & Bernoulli & N'y a-t-il qu'un seul essai ? \\
\hline
Nombre de défauts sur $n$ pièces contrôlées & Binomiale & Répète-t-on $n$ essais indépendants avec la même probabilité de défaut ? \\
\hline
Rang du premier défaut ou du premier succès & Géométrique & Attend-on un premier événement dans une suite d'essais identiques ? \\
\hline
Nombre de pannes ou d'incidents rares sur une durée, une zone ou une longueur & Poisson & Compte-t-on des événements rares sans nombre fixé d'essais au départ ? \\
\hline
\end{tabular}
\end{center}

\section{Erreurs fréquentes}

\begin{itemize}
    \item Choisir une binomiale sans vérifier l'indépendance et la constance de $p$.
    \item Confondre une binomiale avec une géométrique : l'une compte un nombre de succès, l'autre un rang d'attente.
    \item Oublier la différence entre \og nombre de succès \fg{} et \og succès à un essai \fg.
    \item Confondre moyenne observée et espérance théorique.
    \item Utiliser la loi de Poisson pour un petit nombre fixe d'essais.
\end{itemize}

\section{Ce qu'il faut savoir faire}

\begin{itemize}
    \item écrire la loi d'une variable discrète ;
    \item calculer une espérance et une variance ;
    \item reconnaître une Bernoulli, une binomiale, une géométrique ou une Poisson ;
    \item interpréter un résultat en termes de risque ou de coût moyen.
\end{itemize}
