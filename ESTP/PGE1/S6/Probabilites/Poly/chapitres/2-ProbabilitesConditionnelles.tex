% Chapitre 2 : Conditionnement et décision

\section{Pourquoi conditionner ?}

Dans un problème réel, on dispose souvent d'une information supplémentaire :
\begin{itemize}
    \item on sait de quel chantier vient une éprouvette ;
    \item on connaît le résultat d'un capteur de contrôle ;
    \item on sait si la journée était pluvieuse ou non ;
    \item on sait qu'un lot provient d'un fournisseur particulier.
\end{itemize}

Cette information modifie le calcul. C'est exactement le rôle de la probabilité conditionnelle.

\section{Probabilité conditionnelle}

\begin{Def}
Si $A$ et $B$ sont deux événements avec $\mathbb{P}(B)>0$, la \textbf{probabilité conditionnelle} de $A$ sachant $B$ est :
$$
\mathbb{P}(A|B)=\frac{\mathbb{P}(A\cap B)}{\mathbb{P}(B)}.
$$
\end{Def}

\begin{Rmq}
La phrase \og sachant $B$ \fg{} signifie que l'on raisonne \textbf{à l'intérieur du cas où $B$ est déjà réalisé}.
\end{Rmq}

\begin{Ex}
\textbf{Conformité selon le chantier}

On note :
\begin{itemize}
    \item $A$ : \og l'éprouvette est conforme \fg ;
    \item $B$ : \og l'éprouvette provient du chantier A \fg.
\end{itemize}

Alors $\mathbb{P}(A|B)$ est la probabilité d'être conforme \textbf{parmi les éprouvettes du chantier A}.
\end{Ex}

\section{Arbre pondéré et formule des probabilités composées}

\begin{Prop}
Si $A$ et $B$ sont deux événements avec $\mathbb{P}(A)>0$, alors :
$$
\mathbb{P}(A\cap B)=\mathbb{P}(A)\,\mathbb{P}(B|A).
$$
\end{Prop}

\begin{Prop}[Version en chaine]
Pour plusieurs étapes successives :
$$
\mathbb{P}(A_1\cap \cdots \cap A_n)
=\mathbb{P}(A_1)\mathbb{P}(A_2|A_1)\cdots \mathbb{P}(A_n|A_1\cap \cdots \cap A_{n-1}).
$$
\end{Prop}

\begin{Meth}
Un arbre pondéré est utile lorsque :
\begin{itemize}
    \item les situations se découpent en étapes successives ;
    \item on connaît des probabilités conditionnelles ;
    \item on veut calculer une probabilité totale ou une intersection.
\end{itemize}

Pour lire un arbre :
\begin{itemize}
    \item on \textbf{multiplie} le long d'une branche ;
    \item on \textbf{additionne} les branches qui mènent au même événement final.
\end{itemize}
\end{Meth}

\section{Probabilités totales}

\begin{Prop}
Si $(B_1,\dots,B_n)$ forme une partition de l'univers, alors pour tout événement $A$ :
$$
\mathbb{P}(A)=\sum_{i=1}^n \mathbb{P}(A|B_i)\mathbb{P}(B_i).
$$
\end{Prop}

\begin{Ex}
\textbf{Deux fournisseurs}

Un fournisseur $F_1$ livre $70\%$ des pieux et $3\%$ de ses pieux sont defectueux. Un fournisseur $F_2$ livre $30\%$ des pieux et $8\%$ sont defectueux.

Si $D$ est l'evenement \og le pieu est defectueux \fg, alors :
$$
\mathbb{P}(D)=\mathbb{P}(D|F_1)\mathbb{P}(F_1)+\mathbb{P}(D|F_2)\mathbb{P}(F_2)
=0.03\times 0.7 + 0.08\times 0.3 = 0.045.
$$

On attend donc $4.5\%$ de pieces defectueuses au total.
\end{Ex}

\section{Formule de Bayes}

\begin{Prop}[Formule de Bayes]
Si $\mathbb{P}(A)>0$ et $\mathbb{P}(B)>0$, alors :
$$
\mathbb{P}(B|A)=\frac{\mathbb{P}(A|B)\mathbb{P}(B)}{\mathbb{P}(A)}.
$$
\end{Prop}

\begin{Rmq}
Bayes permet \textbf{d'inverser la lecture} :
\begin{itemize}
    \item on connaît souvent $\mathbb{P}(\text{signal}|\text{défaut})$ ;
    \item on cherche plutôt $\mathbb{P}(\text{défaut}|\text{signal})$.
\end{itemize}
\end{Rmq}

\begin{Ex}
\textbf{Capteur de non-conformité}

Un capteur détecte une non-conformité avec une sensibilité de $95\%$ :
$$
\mathbb{P}(S|D)=0.95.
$$
Il produit aussi $4\%$ de faux positifs :
$$
\mathbb{P}(S|\overline{D})=0.04.
$$
Si la proportion de pièces défectueuses vaut $2\%$, alors :
$$
\mathbb{P}(D)=0.02, \qquad \mathbb{P}(\overline{D})=0.98.
$$

La probabilité d'avoir un signal est :
$$
\mathbb{P}(S)=0.95\times 0.02 + 0.04\times 0.98 = 0.0582.
$$

Donc :
$$
\mathbb{P}(D|S)=\frac{0.95\times 0.02}{0.0582}\approx 0.326.
$$

Un signal d'alerte ne signifie donc pas automatiquement qu'il y a défaut ; il faut encore tenir compte de la rareté du défaut.
\end{Ex}

\section{Indépendance : une hypothèse de modélisation}

\begin{Def}
Deux événements $A$ et $B$ sont \textbf{indépendants} si
$$
\mathbb{P}(A\cap B)=\mathbb{P}(A)\mathbb{P}(B).
$$
\end{Def}

\begin{Rmq}
Il ne faut pas confondre :
\begin{itemize}
    \item \textbf{incompatibles} : $A\cap B=\varnothing$ ;
    \item \textbf{indépendants} : connaître $B$ ne change pas la probabilité de $A$.
\end{itemize}
\end{Rmq}

\begin{Ex}
\textbf{Deux éprouvettes}

Deux éprouvettes issues de la même gâchée ne sont pas forcément indépendantes : elles partagent les mêmes conditions de fabrication.

En revanche, deux contrôles effectués sur deux lots indépendants peuvent souvent être modélisés comme indépendants.
\end{Ex}

\begin{Meth}
Pour vérifier l'indépendance de deux événements, on calcule :
\begin{enumerate}
    \item $\mathbb{P}(A)$ ;
    \item $\mathbb{P}(B)$ ;
    \item $\mathbb{P}(A\cap B)$.
\end{enumerate}

On conclut seulement si l'égalité
$$
\mathbb{P}(A\cap B)=\mathbb{P}(A)\mathbb{P}(B)
$$
est vérifiée.
\end{Meth}

\section{Erreurs fréquentes}

\begin{itemize}
    \item Oublier que $\mathbb{P}(A|B)$ ne se calcule que si $\mathbb{P}(B)>0$.
    \item Confondre $\mathbb{P}(A|B)$ et $\mathbb{P}(B|A)$.
    \item Utiliser Bayes alors qu'une simple probabilité totale suffit.
    \item Déclarer l'indépendance sans justification de modèle.
\end{itemize}

\section{Ce qu'il faut savoir faire}

\begin{itemize}
    \item calculer une probabilité conditionnelle ;
    \item lire et exploiter un arbre pondéré ;
    \item utiliser la formule des probabilités totales ;
    \item inverser une condition avec Bayes ;
    \item discuter la pertinence d'une hypothèse d'indépendance.
\end{itemize}
